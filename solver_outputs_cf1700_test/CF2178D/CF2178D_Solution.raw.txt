#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, m;
    cin >> n >> m;
    vector<pair<int, int>> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i].first;
        a[i].second = i + 1;
    }
    sort(a.begin(), a.end());

    if (2 * m > n) {
        cout << "-1\n";
        return;
    }

    vector<pair<int, int>> attacks;

    if (m > 0) {
        // survivors: largest m
        // targets: next m
        // leftovers: smallest n-2m
        int L = n - 2 * m;
        // leftover elves attack the next larger leftover, then the first target
        for (int i = 0; i < L; ++i) {
            int attacker = a[i].second;
            int target;
            if (i + 1 < L) {
                target = a[i + 1].second;
            } else {
                target = a[L].second; // first target
            }
            attacks.emplace_back(attacker, target);
        }
        // survivors attack targets
        for (int i = 0; i < m; ++i) {
            int survivor = a[n - m + i].second;
            int target = a[L + i].second;
            attacks.emplace_back(survivor, target);
        }
    } else {
        // m == 0
        // check if largest can be killed
        long long sum_others = 0;
        for (int i = 0; i < n - 1; ++i) sum_others += a[i].first;
        if (sum_others < a[n - 1].first) {
            cout << "-1\n";
            return;
        }
        // collect elves to attack the largest
        vector<int> to_attack_largest;
        long long cur_sum = 0;
        for (int i = n - 2; i >= 0; --i) {
            if (cur_sum >= a[n - 1].first) break;
            to_attack_largest.push_back(i);
            cur_sum += a[i].first;
        }
        reverse(to_attack_largest.begin(), to_attack_largest.end());
        // remaining smaller elves attack the next larger
        vector<bool> used(n, false);
        for (int idx : to_attack_largest) used[idx] = true;
        int prev = -1;
        for (int i = 0; i < n - 1; ++i) {
            if (used[i]) continue;
            if (prev != -1) {
                attacks.emplace_back(a[prev].second, a[i].second);
            }
            prev = i;
        }
        // now the collected elves attack the largest
        for (int idx : to_attack_largest) {
            attacks.emplace_back(a[idx].second, a[n - 1].second);
        }
    }

    cout << attacks.size() << "\n";
    for (auto &p : attacks) {
        cout << p.first << " " << p.second << "\n";
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
