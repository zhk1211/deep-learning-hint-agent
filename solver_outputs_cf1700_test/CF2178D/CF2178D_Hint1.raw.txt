```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<pair<int, int>> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i].first;
            a[i].second = i + 1;
        }

        if (m == n) {
            cout << -1 << '\n';
            continue;
        }

        sort(a.begin(), a.end());

        if (m == 0) {
            if (n == 2) {
                cout << -1 << '\n';
                continue;
            }
            cout << n - 1 << '\n';
            for (int i = 0; i < n - 1; ++i) {
                cout << a[i].second << ' ' << a[i + 1].second << '\n';
            }
            continue;
        }

        if (m == n - 1) {
            cout << -1 << '\n';
            continue;
        }

        int need = n - m;
        if (need > n - 1) {
            cout << -1 << '\n';
            continue;
        }

        vector<pair<int, int>> attacks;
        vector<bool> attacked(n + 1, false);

        for (int i = 0; i < need; ++i) {
            int x = a[i].second;
            int y = a[i + 1].second;
            attacks.emplace_back(x, y);
            attacked[x] = true;
        }

        cout << attacks.size() << '\n';
        for (auto &p : attacks) {
            cout << p.first << ' ' << p.second << '\n';
        }
    }

    return 0;
}
```
