#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, m;
    cin >> n >> m;
    vector<int> a(n), b(m);
    for (int i = 0; i < n; ++i) cin >> a[i];
    for (int i = 0; i < m; ++i) cin >> b[i];
    
    // Compute LIS of a
    vector<int> lis;
    for (int x : a) {
        auto it = lower_bound(lis.begin(), lis.end(), x);
        if (it == lis.end()) lis.push_back(x);
        else *it = x;
    }
    int L = lis.size();
    
    // We will construct c by placing elements of b in descending order
    // before the first element of a that is <= b_i, or at the end if none.
    // This ensures LIS(c) = L.
    sort(b.begin(), b.end(), greater<int>());
    
    vector<int> c;
    c.reserve(n + m);
    int j = 0; // index in b
    for (int i = 0; i < n; ++i) {
        // Insert all b[j] that are >= a[i] (since b is descending, we check >)
        while (j < m && b[j] >= a[i]) {
            c.push_back(b[j]);
            ++j;
        }
        c.push_back(a[i]);
    }
    // Insert remaining b
    while (j < m) {
        c.push_back(b[j]);
        ++j;
    }
    
    for (int i = 0; i < n + m; ++i) {
        cout << c[i] << " \n"[i == n + m - 1];
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
