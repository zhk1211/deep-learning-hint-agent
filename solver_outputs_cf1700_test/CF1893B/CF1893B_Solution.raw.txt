```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) cin >> a[i];
        vector<int> b(m);
        for (int i = 0; i < m; ++i) cin >> b[i];
        sort(b.begin(), b.end(), greater<int>());

        vector<int> ans;
        ans.reserve(n + m);
        int i = 0, j = 0;
        while (i < n && j < m) {
            if (a[i] >= b[j]) {
                ans.push_back(a[i++]);
            } else {
                ans.push_back(b[j++]);
            }
        }
        while (i < n) ans.push_back(a[i++]);
        while (j < m) ans.push_back(b[j++]);

        for (size_t k = 0; k < ans.size(); ++k) {
            if (k) cout << ' ';
            cout << ans[k];
        }
        cout << '\n';
    }
    return 0;
}
```
