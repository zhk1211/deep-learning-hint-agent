```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    const int MAXN = 100000;
    vector<int> largest_prime(MAXN + 1, 1);
    for (int i = 2; i <= MAXN; ++i) {
        if (largest_prime[i] == 1) {
            for (int j = i; j <= MAXN; j += i) {
                largest_prime[j] = i;
            }
        }
    }

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;

        vector<vector<int>> groups(n + 1);
        for (int x = 2; x <= n; ++x) {
            int lp = largest_prime[x];
            groups[lp].push_back(x);
        }

        vector<int> p(n + 1);
        p[1] = 1;

        for (int prime = 2; prime <= n; ++prime) {
            if (largest_prime[prime] != prime) continue;
            auto& vec = groups[prime];
            if (vec.empty()) continue;
            int sz = vec.size();
            for (int i = 0; i < sz; ++i) {
                p[vec[i]] = vec[(i + 1) % sz];
            }
        }

        for (int i = 1; i <= n; ++i) {
            cout << p[i] << " \n"[i == n];
        }
    }

    return 0;
}
```
