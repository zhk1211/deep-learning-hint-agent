```cpp
#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];

    if (k == 1) {
        cout << "YES\n";
        return;
    }

    vector<int> sorted = a;
    sort(sorted.begin(), sorted.end());
    int x = sorted[k - 2]; // (k-1)-th smallest (0-indexed)

    vector<int> b;
    for (int val : a) {
        if (val <= x) b.push_back(val);
    }

    int L = 0, R = (int)b.size() - 1;
    int deleted = 0;
    while (L <= R) {
        if (b[L] == b[R]) {
            ++L; --R;
        } else if (b[L] == x) {
            ++L;
            ++deleted;
        } else if (b[R] == x) {
            --R;
            ++deleted;
        } else {
            cout << "NO\n";
            return;
        }
    }

    int kept = (int)b.size() - deleted;
    if (kept >= k - 1) {
        cout << "YES\n";
    } else {
        cout << "NO\n";
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) solve();
    return 0;
}
```
