```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXC = 10000000;
int ans[MAXC + 1];

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    fill(ans, ans + MAXC + 1, -1);

    // Precompute sum of divisors for all n up to MAXC
    // We need to find minimal n for each possible sum c
    // Since c <= 1e7, n can be up to 1e7 (actually n <= c-1 for c>1, but we can just go up to MAXC)
    // Use sieve-like approach to compute d(n) and update ans[d(n)] if not set
    vector<int> d(MAXC + 1, 0);
    for (int i = 1; i <= MAXC; ++i) {
        for (int j = i; j <= MAXC; j += i) {
            d[j] += i;
        }
    }

    for (int n = 1; n <= MAXC; ++n) {
        int sum = d[n];
        if (sum <= MAXC && ans[sum] == -1) {
            ans[sum] = n;
        }
    }

    int t;
    cin >> t;
    while (t--) {
        int c;
        cin >> c;
        cout << ans[c] << '\n';
    }

    return 0;
}
```
