#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;
    vector<int> a(n + 2);
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }
    a[0] = 0;
    a[n + 1] = 0;

    long long total = 1LL * n * (n + 1) / 2;
    long long ans = total;
    for (int i = 1; i <= n; ++i) {
        if (a[i] != a[i + 1]) {
            ans += 1LL * i * (n - i);
        }
    }

    while (m--) {
        int i, x;
        cin >> i >> x;
        if (a[i] != x) {
            if (a[i - 1] != a[i] && a[i - 1] == x) {
                ans -= 1LL * (i - 1) * (n - i + 1);
            }
            if (a[i - 1] == a[i] && a[i - 1] != x) {
                ans += 1LL * (i - 1) * (n - i + 1);
            }
            if (a[i] != a[i + 1] && x == a[i + 1]) {
                ans -= 1LL * i * (n - i);
            }
            if (a[i] == a[i + 1] && x != a[i + 1]) {
                ans += 1LL * i * (n - i);
            }
            a[i] = x;
        }
        cout << ans << '\n';
    }

    return 0;
}
