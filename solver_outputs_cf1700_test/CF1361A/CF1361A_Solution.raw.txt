```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < m; ++i) {
        int a, b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }

    vector<int> t(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> t[i];
    }

    // Observation 1: check for edges between same color
    for (int u = 1; u <= n; ++u) {
        for (int v : adj[u]) {
            if (t[u] == t[v]) {
                cout << -1 << '\n';
                return 0;
            }
        }
    }

    // Create permutation sorted by color, then by index
    vector<int> order(n);
    iota(order.begin(), order.end(), 1);
    sort(order.begin(), order.end(), [&](int a, int b) {
        if (t[a] != t[b]) return t[a] < t[b];
        return a < b;
    });

    // Simulate greedy coloring to verify
    vector<int> color(n + 1, 0);
    for (int u : order) {
        // Find the smallest positive integer not present among neighbors already colored
        vector<bool> used(t[u] + 1, false);
        for (int v : adj[u]) {
            if (color[v] != 0 && color[v] <= t[u]) {
                used[color[v]] = true;
            }
        }
        int mex = 1;
        while (mex <= t[u] && used[mex]) ++mex;
        if (mex != t[u]) {
            cout << -1 << '\n';
            return 0;
        }
        color[u] = t[u];
    }

    for (int i = 0; i < n; ++i) {
        cout << order[i] << " \n"[i == n - 1];
    }

    return 0;
}
```
