```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }

    if (n == 1) {
        cout << a[0] << '\n';
        return 0;
    }

    vector<long long> left_req(n), right_req(n);
    left_req[0] = a[0];
    for (int i = 1; i < n; ++i) {
        left_req[i] = max(left_req[i - 1], a[i] - i);
    }
    right_req[n - 1] = a[n - 1] - (n - 1);
    for (int i = n - 2; i >= 0; --i) {
        right_req[i] = max(right_req[i + 1], a[i] - i);
    }

    long long ans = LLONG_MAX;
    for (int i = 0; i < n; ++i) {
        long long cand = a[i];
        if (i > 0) {
            cand = max(cand, left_req[i - 1] + i);
        }
        if (i < n - 1) {
            cand = max(cand, right_req[i + 1] + i);
        }
        ans = min(ans, cand);
    }

    cout << ans << '\n';
    return 0;
}
```
