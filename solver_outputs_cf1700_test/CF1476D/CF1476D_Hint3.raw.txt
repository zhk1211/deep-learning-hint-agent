#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n;
    string s;
    cin >> n >> s;
    
    vector<int> left(n + 1, 0), right(n + 1, 0);
    
    // Compute consecutive alternating segments from left
    left[0] = 1;
    for (int i = 1; i <= n; ++i) {
        if (s[i - 1] == 'L') {
            if (i >= 2 && s[i - 2] == 'R')
                left[i] = left[i - 2] + 2;
            else
                left[i] = 1;
        } else {
            left[i] = 1;
        }
    }
    
    // Compute consecutive alternating segments from right
    right[n] = 1;
    for (int i = n - 1; i >= 0; --i) {
        if (s[i] == 'R') {
            if (i + 2 <= n && s[i + 1] == 'L')
                right[i] = right[i + 2] + 2;
            else
                right[i] = 1;
        } else {
            right[i] = 1;
        }
    }
    
    for (int i = 0; i <= n; ++i) {
        int ans = 1;
        if (i > 0 && s[i - 1] == 'L')
            ans += left[i];
        if (i < n && s[i] == 'R')
            ans += right[i];
        cout << ans << " \n"[i == n];
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
