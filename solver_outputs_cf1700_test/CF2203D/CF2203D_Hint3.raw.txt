```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXV = 2000005;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<int> a(n), b(m);
        for (int i = 0; i < n; ++i) cin >> a[i];
        for (int i = 0; i < m; ++i) cin >> b[i];

        int max_val = 0;
        for (int x : a) max_val = max(max_val, x);
        for (int y : b) max_val = max(max_val, y);

        vector<int> cnt_a(max_val + 1, 0);
        for (int x : a) cnt_a[x]++;

        vector<int> has_div(max_val + 1, 0);
        for (int x = 1; x <= max_val; ++x) {
            if (cnt_a[x] == 0) continue;
            for (int mult = x; mult <= max_val; mult += x) {
                has_div[mult] = 1;
            }
        }

        vector<int> freq_b(max_val + 1, 0);
        for (int y : b) freq_b[y]++;

        int alice_cnt = 0, bob_cnt = 0;
        for (int y = 1; y <= max_val; ++y) {
            if (freq_b[y] == 0) continue;
            bool all_div = true;
            for (int d = 1; d * d <= y; ++d) {
                if (y % d == 0) {
                    if (cnt_a[d] == 0) { all_div = false; break; }
                    int d2 = y / d;
                    if (d2 != d && cnt_a[d2] == 0) { all_div = false; break; }
                }
            }
            if (all_div) {
                alice_cnt += freq_b[y];
            } else if (!has_div[y]) {
                bob_cnt += freq_b[y];
            }
        }

        if (alice_cnt > 0) {
            cout << "Alice\n";
        } else if (bob_cnt > 0) {
            cout << "Bob\n";
        } else {
            cout << "Alice\n";
        }
    }
    return 0;
}
```
