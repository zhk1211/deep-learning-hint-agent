#include <bits/stdc++.h>
using namespace std;

using ll = long long;

void solve() {
    int n;
    ll c;
    cin >> n >> c;
    vector<ll> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    // We will try to connect everything to node 1 (index 0).
    // We need to process other nodes in order of a[i] - (i+1)*c descending.
    vector<pair<ll, int>> v;
    for (int i = 1; i < n; ++i) {
        v.push_back({a[i] - (i + 1) * c, i});
    }
    sort(v.begin(), v.end(), greater<pair<ll, int>>());
    ll sum = a[0];
    for (auto &p : v) {
        int idx = p.second;
        if (sum + a[idx] >= (ll)(idx + 1) * c) {
            sum += a[idx];
        } else {
            cout << "No\n";
            return;
        }
    }
    cout << "Yes\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
