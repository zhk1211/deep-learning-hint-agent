#include <bits/stdc++.h>
using namespace std;
using i64 = long long;

void solve() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }

    vector<pair<int, int>> st; // (value, index when added)
    i64 cur = 0; // sum of f for all prefixes of current suffix
    i64 ans = 0;

    for (int i = n - 1; i >= 0; --i) {
        int v = a[i];
        int idx = i;
        while (!st.empty() && st.back().first < v) {
            auto [val, pos] = st.back();
            st.pop_back();
            int nxt = st.empty() ? n : st.back().second;
            cur -= (i64)(val + 1) * (nxt - pos);
            idx = nxt;
        }
        if (!st.empty() && st.back().first == v) {
            idx = st.back().second;
            st.pop_back();
        }
        st.emplace_back(v, idx);
        int nxt = st.size() > 1 ? st[st.size() - 2].second : n;
        cur += (i64)(v + 1) * (nxt - idx);
        ans += cur;
    }

    cout << ans << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
