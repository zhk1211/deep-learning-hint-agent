#include <bits/stdc++.h>
using namespace std;

using ll = long long;

void solve() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];

    vector<int> dp(n);
    vector<int> stk;
    stk.reserve(n);
    ll ans = 0;
    ll sum_dp = 0;

    for (int i = 0; i < n; ++i) {
        while (!stk.empty() && a[stk.back()] + (i - stk.back()) <= a[i]) {
            int j = stk.back();
            stk.pop_back();
            int prev = stk.empty() ? -1 : stk.back();
            sum_dp -= dp[j];
            if (prev != -1) {
                dp[prev] = dp[prev] + dp[j] - (a[j] + (j - prev) - a[prev]);
                sum_dp += dp[prev];
            }
        }
        int prev = stk.empty() ? -1 : stk.back();
        if (prev == -1) {
            dp[i] = max(0, a[i] - i);
        } else {
            dp[i] = dp[prev] + max(0, a[i] - (a[prev] + (i - prev)));
        }
        sum_dp += dp[i];
        stk.push_back(i);
        ans += sum_dp;
    }
    cout << ans << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) solve();
    return 0;
}
