#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n;
    cin >> n;
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    vector<int> depth(n + 1), max_depth(n + 1);
    vector<int> order;
    function<void(int, int)> dfs = [&](int u, int p) {
        order.push_back(u);
        max_depth[u] = depth[u];
        for (int v : adj[u]) {
            if (v == p) continue;
            depth[v] = depth[u] + 1;
            dfs(v, u);
            max_depth[u] = max(max_depth[u], max_depth[v]);
        }
    };
    depth[1] = 0;
    dfs(1, 0);
    
    vector<int> diff(n + 2, 0);
    for (int u = 2; u <= n; u++) {
        if (adj[u].size() == 1) {
            int a = depth[u];
            int b = max_depth[u];
            diff[a]++;
            diff[b + 1]--;
        }
    }
    
    int best = 0, cur = 0;
    for (int d = 0; d <= n; d++) {
        cur += diff[d];
        best = max(best, cur);
    }
    
    int leaves = 0;
    for (int u = 2; u <= n; u++) {
        if (adj[u].size() == 1) leaves++;
    }
    
    cout << leaves - best << "\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
