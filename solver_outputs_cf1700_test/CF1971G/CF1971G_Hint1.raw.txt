```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }

        // Group indices by the prefix ignoring the last two bits
        // Two numbers can be swapped if they differ only in the last two bits
        // i.e., a_i >> 2 == a_j >> 2
        map<int, vector<int>> groups;
        for (int i = 0; i < n; ++i) {
            groups[a[i] >> 2].push_back(i);
        }

        vector<int> ans(n);
        for (auto& [key, indices] : groups) {
            // Collect values in this group
            vector<int> vals;
            for (int idx : indices) {
                vals.push_back(a[idx]);
            }
            // Sort values to place smallest possible at smallest indices
            sort(vals.begin(), vals.end());
            // Sort indices to fill lexicographically smallest
            sort(indices.begin(), indices.end());
            for (size_t i = 0; i < indices.size(); ++i) {
                ans[indices[i]] = vals[i];
            }
        }

        for (int i = 0; i < n; ++i) {
            cout << ans[i] << " \n"[i == n - 1];
        }
    }

    return 0;
}
```
