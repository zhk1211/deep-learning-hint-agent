```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> p(n + 1);
        for (int i = 2; i <= n; ++i) {
            cin >> p[i];
        }
        vector<long long> l(n + 1), r(n + 1);
        for (int i = 1; i <= n; ++i) {
            cin >> l[i] >> r[i];
        }

        vector<vector<int>> children(n + 1);
        for (int i = 2; i <= n; ++i) {
            children[p[i]].push_back(i);
        }

        int ans = 0;
        vector<long long> sum(n + 1, 0);

        function<void(int)> dfs = [&](int u) {
            long long total = 0;
            for (int v : children[u]) {
                dfs(v);
                total += sum[v];
            }
            if (total < l[u]) {
                ++ans;
                sum[u] = r[u];
            } else {
                sum[u] = min(total, r[u]);
            }
        };

        dfs(1);
        cout << ans << '\n';
    }

    return 0;
}
```
