#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        
        // Method 1: q_i = (i-1) XOR (i-2) for i>=2, q_1 = 0
        for (int i = 0; i < n; i++) {
            if (i == 0) {
                cout << 0 << endl;
            } else {
                cout << (i ^ (i - 1)) << endl;
            }
            int r;
            cin >> r;
            if (r == 1) break;
            if (r == -1) return 0;
        }
    }
    return 0;
}
