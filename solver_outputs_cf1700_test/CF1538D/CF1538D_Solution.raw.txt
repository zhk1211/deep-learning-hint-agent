#include <bits/stdc++.h>
using namespace std;

int count_prime_factors(int x) {
    int cnt = 0;
    for (int i = 2; i * i <= x; ++i) {
        while (x % i == 0) {
            ++cnt;
            x /= i;
        }
    }
    if (x > 1) ++cnt;
    return cnt;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int a, b, k;
        cin >> a >> b >> k;
        
        int n = count_prime_factors(a) + count_prime_factors(b);
        int m;
        if (a == b) {
            m = 0;
        } else if (gcd(a, b) == a || gcd(a, b) == b) {
            m = 1;
        } else {
            m = 2;
        }
        
        bool ok = false;
        if (k == 1) {
            if (m == 1 && k <= n) ok = true;
        } else {
            if (m <= k && k <= n) ok = true;
        }
        
        cout << (ok ? "YES" : "NO") << '\n';
    }
    return 0;
}
