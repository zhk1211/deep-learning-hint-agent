#include <bits/stdc++.h>
using namespace std;

bool is_prime(int x) {
    if (x < 2) return false;
    for (int i = 2; i * i <= x; ++i) {
        if (x % i == 0) return false;
    }
    return true;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;

        int L = n / 3;
        int R = (2 * n + 2) / 3; // ceil(2n/3)
        int p = -1;
        for (int x = L; x <= R; ++x) {
            if (is_prime(x)) {
                p = x;
                break;
            }
        }
        // Bertrand's postulate guarantees such p exists for n>=2

        vector<int> perm;
        vector<bool> used(n + 1, false);

        int low = p - 1, high = p + 1;
        perm.push_back(p);
        used[p] = true;

        bool turn_low = true;
        while ((int)perm.size() < n) {
            if (turn_low) {
                while (low >= 1 && used[low]) --low;
                if (low >= 1) {
                    perm.push_back(low);
                    used[low] = true;
                }
            } else {
                while (high <= n && used[high]) ++high;
                if (high <= n) {
                    perm.push_back(high);
                    used[high] = true;
                }
            }
            turn_low = !turn_low;
        }

        for (int i = 0; i < n; ++i) {
            cout << perm[i] << " \n"[i == n - 1];
        }
    }
    return 0;
}
