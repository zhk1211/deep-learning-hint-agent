#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> p(n);
        if (n == 2) {
            p = {2, 1};
        } else if (n == 3) {
            p = {2, 1, 3};
        } else {
            p[0] = 2;
            p[1] = 1;
            p[2] = 3;
            for (int i = 3; i < n; ++i) {
                p[i] = i + 1;
            }
        }
        for (int i = 0; i < n; ++i) {
            cout << p[i] << " \n"[i == n - 1];
        }
    }
    return 0;
}
