#include <bits/stdc++.h>
using namespace std;

const int MAXA = 100000;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }
        sort(a.begin(), a.end());

        vector<int> cnt(m + 1, 0);
        int covered = 0;
        int ans = INT_MAX;
        int r = 0;
        for (int l = 0; l < n; ++l) {
            while (r < n && covered < m) {
                int val = a[r];
                for (int d = 1; d * d <= val && d <= m; ++d) {
                    if (val % d == 0) {
                        if (++cnt[d] == 1) covered++;
                        int d2 = val / d;
                        if (d2 != d && d2 <= m) {
                            if (++cnt[d2] == 1) covered++;
                        }
                    }
                }
                ++r;
            }
            if (covered == m) {
                ans = min(ans, a[r - 1] - a[l]);
            }
            int val = a[l];
            for (int d = 1; d * d <= val && d <= m; ++d) {
                if (val % d == 0) {
                    if (--cnt[d] == 0) covered--;
                    int d2 = val / d;
                    if (d2 != d && d2 <= m) {
                        if (--cnt[d2] == 0) covered--;
                    }
                }
            }
        }
        if (ans == INT_MAX) ans = -1;
        cout << ans << '\n';
    }
    return 0;
}
