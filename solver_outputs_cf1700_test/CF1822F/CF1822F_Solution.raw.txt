#include <bits/stdc++.h>
using namespace std;

using ll = long long;

void solve() {
    int n;
    ll k, c;
    cin >> n >> k >> c;
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < n - 1; ++i) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    vector<int> depth(n + 1);
    vector<ll> down1(n + 1, 0), down2(n + 1, 0);
    vector<int> best_child(n + 1, 0);

    function<void(int, int)> dfs1 = [&](int u, int p) {
        for (int v : adj[u]) {
            if (v == p) continue;
            depth[v] = depth[u] + 1;
            dfs1(v, u);
            ll val = down1[v] + 1;
            if (val > down1[u]) {
                down2[u] = down1[u];
                down1[u] = val;
                best_child[u] = v;
            } else if (val > down2[u]) {
                down2[u] = val;
            }
        }
    };

    depth[1] = 0;
    dfs1(1, 0);

    vector<ll> up(n + 1, 0);
    function<void(int, int)> dfs2 = [&](int u, int p) {
        for (int v : adj[u]) {
            if (v == p) continue;
            if (best_child[u] == v) {
                up[v] = max(up[u], down2[u]) + 1;
            } else {
                up[v] = max(up[u], down1[u]) + 1;
            }
            dfs2(v, u);
        }
    };

    dfs2(1, 0);

    ll ans = 0;
    for (int v = 1; v <= n; ++v) {
        ll dist = max(down1[v], up[v]);
        ll profit = k * dist - c * depth[v];
        ans = max(ans, profit);
    }
    cout << ans << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
