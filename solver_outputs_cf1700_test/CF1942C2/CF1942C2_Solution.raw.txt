#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        long long n;
        int x, y;
        cin >> n >> x >> y;
        vector<int> a(x);
        for (int i = 0; i < x; ++i) {
            cin >> a[i];
        }
        sort(a.begin(), a.end());

        long long ans = x - 2;
        vector<int> gaps;
        for (int i = 0; i < x; ++i) {
            int nxt = (i + 1 == x) ? a[0] + n : a[i + 1];
            int g = nxt - a[i] - 1;
            if (g == 1) {
                ans += 1;
            } else if (g > 1) {
                gaps.push_back(g);
            }
        }

        sort(gaps.begin(), gaps.end(), [](int g1, int g2) {
            if ((g1 % 2) != (g2 % 2)) return (g1 % 2) < (g2 % 2);
            return g1 < g2;
        });

        for (int g : gaps) {
            if (y == 0) break;
            if (g % 2 == 1) {
                int take = min(y, g / 2);
                ans += take * 2;
                y -= take;
                if (take == g / 2) {
                    ans += 1;
                }
            } else {
                int take = min(y, g / 2);
                ans += take * 2;
                y -= take;
            }
        }

        cout << ans << '\n';
    }
    return 0;
}
