#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, k;
    cin >> n >> k;
    if (k < n || k > 2 * n - 1) {
        cout << "NO\n";
        return;
    }
    if (n == 1) {
        if (k == 1) {
            cout << "YES\n1 1\n";
        } else {
            cout << "NO\n";
        }
        return;
    }
    if (k == 2 * n) {
        cout << "NO\n";
        return;
    }
    cout << "YES\n";
    vector<int> a(2 * n);
    if (k == n) {
        for (int i = 0; i < n; ++i) {
            a[2 * i] = a[2 * i + 1] = i + 1;
        }
    } else {
        int d = k - n;
        for (int i = 0; i < n; ++i) {
            a[i] = i + 1;
        }
        for (int i = 0; i < n; ++i) {
            a[n + i] = i + 1;
        }
        if (d == 1) {
            swap(a[0], a[1]);
        } else {
            int pos = d;
            int val = a[pos];
            for (int i = pos; i > 0; --i) {
                a[i] = a[i - 1];
            }
            a[0] = val;
        }
    }
    for (int i = 0; i < 2 * n; ++i) {
        cout << a[i] << " \n"[i == 2 * n - 1];
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
