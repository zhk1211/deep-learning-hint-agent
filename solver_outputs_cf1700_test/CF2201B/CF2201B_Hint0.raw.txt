#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, k;
    cin >> n >> k;
    
    // The algorithm always takes at least n turns (when all pairs are matched immediately)
    // and at most 2n-1 turns (when we keep flipping new cards until the end)
    if (k < n || k > 2 * n - 1) {
        cout << "NO\n";
        return;
    }
    
    // We need to construct a sequence of length 2n with each number 1..n appearing twice
    // such that the greedy algorithm takes exactly k turns.
    
    // Key insight: The number of turns equals n + (number of "failed" first flips)
    // A "failed" first flip happens when we flip a new card as first, and its pair is not among previously seen cards.
    // We can control this by placing pairs strategically.
    
    // Let's construct the sequence as follows:
    // We'll have some prefix of "singletons" that cause failed first flips,
    // followed by pairs that will be matched immediately or after some delay.
    
    // Let f = k - n (number of failed first flips). 0 <= f <= n-1.
    int f = k - n;
    
    vector<int> ans;
    ans.reserve(2 * n);
    
    // We'll use numbers 1..f+1 for the "chain" that causes f failed flips,
    // and the remaining numbers for immediate pairs.
    
    // The chain: place first occurrences of 1,2,...,f+1 in order,
    // then place second occurrence of 1, then second occurrence of 2, ..., second occurrence of f+1.
    // This creates exactly f failed first flips.
    // Let's verify: 
    // Turn 1: flip 1 (new), then flip 2 (new) -> fail (f=1 used)
    // Turn 2: flip 1 (seen before? no, it was flipped back), actually algorithm:
    // After turn 1, we have seen 1 and 2. Turn 2: first card never flipped is 1? No, 1 was flipped.
    // Actually the algorithm: "If there are two cards that you have flipped previously and have the same number, flip those two cards."
    // After turn 1, we have flipped 1 and 2, they are different, so no pair. Then "Otherwise, flip the first card that you have never flipped so far as the first one."
    // The first never flipped card is the third card (which is 3 if f>=2, or second occurrence of 1 if f=1).
    // Wait, this needs careful analysis.
    
    // Let's design a simpler construction:
    // We want exactly f times where we flip a new card and its pair is not yet seen.
    // We can achieve this by having f+1 numbers that appear as: first occurrences interleaved with other numbers.
    
    // Known construction from editorial:
    // Place numbers 1,2,...,f+1 as first occurrences.
    // Then place the second occurrence of 1.
    // Then place numbers f+2, f+3, ..., n as pairs (each number twice in a row).
    // Then place second occurrences of 2,3,...,f+1.
    // This gives exactly k = n + f turns.
    
    // Let's implement this:
    // First part: 1, 2, 3, ..., f+1
    for (int i = 1; i <= f + 1; ++i) {
        ans.push_back(i);
    }
    // Second part: second occurrence of 1
    ans.push_back(1);
    // Third part: pairs of remaining numbers
    for (int i = f + 2; i <= n; ++i) {
        ans.push_back(i);
        ans.push_back(i);
    }
    // Fourth part: second occurrences of 2,3,...,f+1
    for (int i = 2; i <= f + 1; ++i) {
        ans.push_back(i);
    }
    
    // Check length
    assert((int)ans.size() == 2 * n);
    
    cout << "YES\n";
    for (int i = 0; i < 2 * n; ++i) {
        cout << ans[i] << " \n"[i == 2 * n - 1];
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
