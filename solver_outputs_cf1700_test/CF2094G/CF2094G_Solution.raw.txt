```cpp
#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int q;
        cin >> q;
        deque<ll> dq;
        ll score = 0, sum = 0, sz = 0;
        bool rev = false;

        auto get_score = [&]() -> ll {
            if (!rev) return score;
            // rscore = (sz + 1) * sum - score
            return (sz + 1) * sum - score;
        };

        while (q--) {
            int op;
            cin >> op;
            if (op == 1) {
                // cyclic shift
                if (sz == 0) {
                    cout << 0 << "\n";
                    continue;
                }
                if (!rev) {
                    // pop back, push front
                    ll back_val = dq.back();
                    dq.pop_back();
                    // score decreases by back_val * sz
                    score -= back_val * sz;
                    // push front
                    dq.push_front(back_val);
                    // score increases by sum
                    score += sum;
                } else {
                    // reversed: pop front, push back
                    ll front_val = dq.front();
                    dq.pop_front();
                    // score decreases by sum (since front moves out)
                    score -= sum;
                    // push back
                    dq.push_back(front_val);
                    // score increases by front_val * sz
                    score += front_val * sz;
                }
                cout << get_score() << "\n";
            } else if (op == 2) {
                // reverse
                rev = !rev;
                cout << get_score() << "\n";
            } else if (op == 3) {
                ll k;
                cin >> k;
                sz++;
                sum += k;
                if (!rev) {
                    dq.push_back(k);
                    score += k * sz;
                } else {
                    dq.push_front(k);
                    // rscore increases by sum (new sum)
                    // score = (sz+1)*sum - rscore, so score increases by k*sz
                    score += k * sz;
                }
                cout << get_score() << "\n";
            }
        }
    }
    return 0;
}
```
