#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n;
    cin >> n;
    vector<int> p(n + 1, 0);
    vector<bool> visited(n + 1, false);
    
    for (int i = 1; i <= n; ++i) {
        if (!visited[i]) {
            vector<int> cycle;
            int cur = i;
            while (true) {
                cout << "? " << i << endl;
                int val;
                cin >> val;
                if (val == 0) exit(0);
                cycle.push_back(val);
                if (val == i) break;
            }
            // The cycle is in order: first element is q_i after first query,
            // second is q_i after second query, etc.
            // The permutation p maps each element to the next in the cycle.
            for (int j = 0; j < (int)cycle.size(); ++j) {
                int from = cycle[j];
                int to = cycle[(j + 1) % cycle.size()];
                p[from] = to;
                visited[from] = true;
            }
        }
    }
    
    cout << "!";
    for (int i = 1; i <= n; ++i) {
        cout << " " << p[i];
    }
    cout << endl;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
