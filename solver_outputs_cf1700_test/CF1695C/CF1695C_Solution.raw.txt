```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<vector<int>> a(n, vector<int>(m));
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < m; ++j) {
                cin >> a[i][j];
            }
        }
        
        if ((n + m - 1) % 2 != 0) {
            cout << "NO\n";
            continue;
        }
        
        vector<vector<int>> min_dp(n, vector<int>(m));
        vector<vector<int>> max_dp(n, vector<int>(m));
        
        min_dp[0][0] = max_dp[0][0] = a[0][0];
        
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < m; ++j) {
                if (i == 0 && j == 0) continue;
                int min_val = INT_MAX, max_val = INT_MIN;
                if (i > 0) {
                    min_val = min(min_val, min_dp[i-1][j]);
                    max_val = max(max_val, max_dp[i-1][j]);
                }
                if (j > 0) {
                    min_val = min(min_val, min_dp[i][j-1]);
                    max_val = max(max_val, max_dp[i][j-1]);
                }
                min_dp[i][j] = min_val + a[i][j];
                max_dp[i][j] = max_val + a[i][j];
            }
        }
        
        if (min_dp[n-1][m-1] <= 0 && 0 <= max_dp[n-1][m-1]) {
            cout << "YES\n";
        } else {
            cout << "NO\n";
        }
    }
    
    return 0;
}
```
