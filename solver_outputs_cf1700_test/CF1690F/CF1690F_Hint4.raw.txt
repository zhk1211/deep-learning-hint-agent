```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        vector<int> p(n);
        for (int i = 0; i < n; ++i) {
            cin >> p[i];
            --p[i];
        }

        vector<bool> visited(n, false);
        long long ans = 1;

        for (int i = 0; i < n; ++i) {
            if (!visited[i]) {
                vector<int> cycle;
                int cur = i;
                while (!visited[cur]) {
                    visited[cur] = true;
                    cycle.push_back(cur);
                    cur = p[cur];
                }

                int len = cycle.size();
                string orig, cur_str;
                for (int idx : cycle) orig += s[idx];
                cur_str = orig;

                long long min_shift = len;
                for (int shift = 1; shift <= len; ++shift) {
                    rotate(cur_str.begin(), cur_str.begin() + 1, cur_str.end());
                    if (cur_str == orig) {
                        min_shift = shift;
                        break;
                    }
                }

                ans = lcm(ans, min_shift);
            }
        }

        cout << ans << '\n';
    }

    return 0;
}
```
