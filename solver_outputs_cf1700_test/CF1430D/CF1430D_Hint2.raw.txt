```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n;
        string s;
        cin >> n >> s;
        
        vector<int> blocks;
        int cnt = 1;
        for (int i = 1; i < n; i++) {
            if (s[i] == s[i-1]) {
                cnt++;
            } else {
                blocks.push_back(cnt);
                cnt = 1;
            }
        }
        blocks.push_back(cnt);
        
        int m = blocks.size();
        int ans = 0;
        int j = 0;
        for (int i = 0; i < m; i++) {
            if (blocks[i] > 1) {
                ans++;
            } else {
                while (j < m && (j < i || blocks[j] == 1)) {
                    j++;
                }
                if (j < m) {
                    blocks[j]--;
                    ans++;
                } else {
                    ans += (m - i + 1) / 2;
                    break;
                }
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
```
