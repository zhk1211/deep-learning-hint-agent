```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<vector<int>> adj(n + 1);
        for (int i = 0; i < n - 1; ++i) {
            int u, v;
            cin >> u >> v;
            adj[u].push_back(v);
            adj[v].push_back(u);
        }
        string s;
        cin >> s;
        s = " " + s;
        vector<int> parent(n + 1, 0);
        vector<int> order;
        order.reserve(n);
        queue<int> q;
        q.push(1);
        parent[1] = -1;
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            order.push_back(u);
            for (int v : adj[u]) {
                if (v != parent[u]) {
                    parent[v] = u;
                    q.push(v);
                }
            }
        }
        vector<int> leaf(n + 1, 0);
        for (int u = 2; u <= n; ++u) {
            if (adj[u].size() == 1) leaf[u] = 1;
        }
        int cnt_q = 0;
        for (int i = 1; i <= n; ++i) {
            if (s[i] == '?') cnt_q++;
        }
        if (s[1] != '?') {
            int base = 0;
            for (int u = 2; u <= n; ++u) {
                if (leaf[u]) {
                    int cur = u;
                    char last = s[cur];
                    while (parent[cur] != -1) {
                        cur = parent[cur];
                        if (s[cur] != '?') {
                            if (s[cur] != last) {
                                base++;
                            }
                            break;
                        }
                    }
                }
            }
            int cnt_q_leaf = 0;
            for (int u = 2; u <= n; ++u) {
                if (leaf[u] && s[u] == '?') cnt_q_leaf++;
            }
            int other_q = cnt_q - cnt_q_leaf;
            int ans = base + (cnt_q_leaf + 1) / 2;
            if (other_q % 2 == 1) {
                if (s[1] == '0') ans = max(ans, base + cnt_q_leaf / 2 + 1);
                else ans = max(ans, base + (cnt_q_leaf + 1) / 2);
            }
            cout << ans << '\n';
        } else {
            int base0 = 0, base1 = 0;
            for (int u = 2; u <= n; ++u) {
                if (leaf[u]) {
                    int cur = u;
                    char last = s[cur];
                    while (parent[cur] != -1) {
                        cur = parent[cur];
                        if (s[cur] != '?') {
                            if (s[cur] != last) {
                                if (s[cur] == '0') base0++;
                                else base1++;
                            }
                            break;
                        }
                    }
                }
            }
            int cnt_q_leaf = 0;
            for (int u = 2; u <= n; ++u) {
                if (leaf[u] && s[u] == '?') cnt_q_leaf++;
            }
            int other_q = cnt_q - cnt_q_leaf - 1;
            int ans0 = base0 + (cnt_q_leaf + 1) / 2;
            int ans1 = base1 + (cnt_q_leaf + 1) / 2;
            if (other_q % 2 == 1) {
                ans0 = max(ans0, base0 + cnt_q_leaf / 2 + 1);
                ans1 = max(ans1, base1 + cnt_q_leaf / 2 + 1);
            }
            cout << max(ans0, ans1) << '\n';
        }
    }
    return 0;
}
```
