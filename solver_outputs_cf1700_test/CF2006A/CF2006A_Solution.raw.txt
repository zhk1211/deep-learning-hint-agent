#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n;
    cin >> n;
    vector<int> deg(n + 1, 0);
    for (int i = 0; i < n - 1; ++i) {
        int u, v;
        cin >> u >> v;
        deg[u]++;
        deg[v]++;
    }
    string s;
    cin >> s;
    s = " " + s;
    
    int x = 0, y = 0, z = 0, w = 0;
    for (int i = 2; i <= n; ++i) {
        if (deg[i] == 1) {
            if (s[i] == '?') z++;
            else if (s[i] == '0') x++;
            else y++;
        } else if (s[i] == '?') {
            w++;
        }
    }
    
    if (s[1] == '0') {
        cout << y + (z + 1) / 2 << '\n';
    } else if (s[1] == '1') {
        cout << x + (z + 1) / 2 << '\n';
    } else {
        int add = (x == y) ? (w % 2) : 0;
        cout << max(x, y) + (z + add) / 2 << '\n';
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
