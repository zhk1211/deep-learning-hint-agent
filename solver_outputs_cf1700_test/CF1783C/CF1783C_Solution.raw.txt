```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }

        vector<int> sorted_a = a;
        sort(sorted_a.begin(), sorted_a.end());

        int x = 0;
        long long sum = 0;
        while (x < n && sum + sorted_a[x] <= m) {
            sum += sorted_a[x];
            ++x;
        }

        if (x == n) {
            cout << 1 << '\n';
            continue;
        }

        int place;
        if (x > 0 && a[x] <= sorted_a[x - 1]) {
            place = n - x;
        } else {
            long long new_sum = sum;
            if (x > 0) {
                new_sum -= sorted_a[x - 1];
                new_sum += a[x];
            } else {
                new_sum = a[x];
            }
            if (new_sum <= m) {
                place = n - x;
            } else {
                place = n - x + 1;
            }
        }

        cout << place << '\n';
    }

    return 0;
}
```
