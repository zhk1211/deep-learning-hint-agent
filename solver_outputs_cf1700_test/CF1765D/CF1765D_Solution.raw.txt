```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    long long m;
    cin >> n >> m;
    vector<long long> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());

    long long total_sum = accumulate(a.begin(), a.end(), 0LL);
    long long base_time = n + total_sum;

    auto can = [&](int k) -> bool {
        int sz = n - k;
        for (int i = 0; i < sz / 2; ++i) {
            if (a[i] + a[sz - 1 - i] > m) {
                return false;
            }
        }
        return true;
    };

    int lo = 0, hi = n - 1, best_k = n - 1;
    while (lo <= hi) {
        int mid = (lo + hi) / 2;
        if (can(mid)) {
            best_k = mid;
            hi = mid - 1;
        } else {
            lo = mid + 1;
        }
    }

    long long ans = base_time - (n - 1 - best_k);
    cout << ans << "\n";
    return 0;
}
```
