#include <bits/stdc++.h>
using namespace std;

using ll = long long;

ll solve_case(const vector<ll>& h) {
    ll mx = *max_element(h.begin(), h.end());
    ll ans = LLONG_MAX;
    for (ll target : {mx, mx + 1}) {
        ll odd_needed = 0, even_needed = 0;
        for (ll x : h) {
            ll diff = target - x;
            even_needed += diff / 2;
            odd_needed += diff % 2;
        }
        // Try to balance: we can replace two odd days with one even day
        // Actually we want to minimize max(2*odd-1, 2*even) or something similar
        // Standard approach: binary search on days
        auto can = [&](ll days) -> bool {
            ll odd_days = (days + 1) / 2;
            ll even_days = days / 2;
            ll need_odd = odd_needed;
            ll need_even = even_needed;
            // Use odd days first for odd increments
            ll use_odd = min(need_odd, odd_days);
            need_odd -= use_odd;
            odd_days -= use_odd;
            // Remaining odd days can be used as even (by skipping one day? No, odd day gives +1, even day gives +2)
            // Actually we can convert odd day to even by not using it? No, we can use odd day to give +1, but if we need +2 we can't.
            // Better: we can use even days to cover odd needs by using two even days? No.
            // Standard trick: we can use even days to cover odd needs by using one even day to give +2, but we need +1, so we waste 1.
            // So we can satisfy odd_needed with odd_days, and if we lack odd_days, we can use even_days to cover odd_needed by using 2 even_days per odd_needed? Actually one even day gives +2, so to cover one odd need we need one even day and waste 1.
            // But we can also convert odd_days to even by using two odd days to give +2.
            // So we need to check if total increments possible >= total needed, with constraints.
            // Total increments possible = odd_days + 2*even_days.
            // But we also need to satisfy odd_needed with odd_days or by using even_days (each even day can cover one odd need but wastes 1).
            // Actually we can think: we need to assign increments to days. Each odd day gives +1, each even day gives +2.
            // We need to cover odd_needed (which are +1 increments) and even_needed (which are +2 increments).
            // We can use odd days for +1, even days for +2.
            // If we run out of odd days, we can use even days to give +1 by using one even day and wasting 1 (i.e., we use the +2 as +1 and lose 1).
            // If we run out of even days, we can use two odd days to give +2.
            // So we need to check if we can satisfy:
            // odd_needed <= odd_days + even_days (each even day can cover one odd need)
            // even_needed <= even_days + odd_days/2 (each two odd days can cover one even need)
            // But also total increments: odd_days + 2*even_days >= odd_needed + 2*even_needed.
            // However the first two conditions are sufficient if total increments condition holds? Not exactly.
            // Let's just simulate greedily: use odd days for odd needs first, then use remaining odd days for even needs (two per even need), then use even days for even needs, then use even days for odd needs.
            ll rem_odd = odd_days, rem_even = even_days;
            ll need_o = odd_needed, need_e = even_needed;
            // Use odd days for odd needs
            ll take = min(need_o, rem_odd);
            need_o -= take;
            rem_odd -= take;
            // Use remaining odd days for even needs (two odd days per even need)
            take = min(need_e, rem_odd / 2);
            need_e -= take;
            rem_odd -= take * 2;
            // Use even days for even needs
            take = min(need_e, rem_even);
            need_e -= take;
            rem_even -= take;
            // Use even days for odd needs (one even day per odd need)
            take = min(need_o, rem_even);
            need_o -= take;
            rem_even -= take;
            // Use remaining even days for even needs? Already done.
            // If both needs are zero, possible
            return need_o == 0 && need_e == 0;
        };
        ll low = 0, high = 1e18;
        while (low < high) {
            ll mid = (low + high) / 2;
            if (can(mid)) high = mid;
            else low = mid + 1;
        }
        ans = min(ans, low);
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<ll> h(n);
        for (int i = 0; i < n; ++i) cin >> h[i];
        cout << solve_case(h) << '\n';
    }
    return 0;
}
