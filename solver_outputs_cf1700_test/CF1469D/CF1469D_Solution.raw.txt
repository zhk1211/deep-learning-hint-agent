#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<pair<int, int>> ops;
        int cur = n;
        while (cur > 2) {
            int y = (int)ceil(sqrt((double)cur));
            for (int i = y + 1; i < cur; ++i) {
                ops.emplace_back(i, cur);
            }
            ops.emplace_back(cur, y);
            ops.emplace_back(cur, y);
            cur = y;
        }
        cout << ops.size() << '\n';
        for (auto [x, y] : ops) {
            cout << x << ' ' << y << '\n';
        }
    }
    return 0;
}
