#include <bits/stdc++.h>
using namespace std;
using ll = long long;

void solve() {
    int n, q;
    cin >> n >> q;
    vector<ll> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    
    vector<ll> pref(2 * n + 1, 0);
    for (int i = 0; i < 2 * n; ++i) {
        pref[i + 1] = pref[i] + a[i % n];
    }
    
    ll total = pref[n];
    
    auto get_sum = [&](ll l, ll r) -> ll {
        if (l > r) return 0;
        ll res = 0;
        ll block_l = (l - 1) / n;
        ll block_r = (r - 1) / n;
        ll idx_l = (l - 1) % n;
        ll idx_r = (r - 1) % n;
        
        if (block_l == block_r) {
            res = pref[block_l + idx_r + 1] - pref[block_l + idx_l];
        } else {
            res += pref[block_l + n] - pref[block_l + idx_l];
            res += pref[block_r + idx_r + 1] - pref[block_r];
            if (block_l + 1 < block_r) {
                res += total * (block_r - block_l - 1);
            }
        }
        return res;
    };
    
    while (q--) {
        ll l, r;
        cin >> l >> r;
        cout << get_sum(l, r) << '\n';
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
