#include <bits/stdc++.h>
using namespace std;
using i64 = long long;

void solve() {
    int n, q;
    cin >> n >> q;
    vector<i64> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    i64 total = accumulate(a.begin(), a.end(), 0LL);
    vector<i64> pref(2 * n + 1, 0);
    for (int i = 0; i < 2 * n; ++i) {
        pref[i + 1] = pref[i] + a[i % n];
    }
    auto range_sum = [&](int l, int r) -> i64 {
        if (l > r) return 0;
        return pref[r + 1] - pref[l];
    };
    auto query = [&](i64 idx) -> i64 {
        i64 block = (idx - 1) / n;
        i64 pos = (idx - 1) % n;
        i64 ans = block * total;
        i64 start = block % n;
        i64 end = (start + pos) % n;
        if (start <= end) {
            ans += range_sum(start, end);
        } else {
            ans += range_sum(start, n - 1) + range_sum(0, end);
        }
        return ans;
    };
    while (q--) {
        i64 l, r;
        cin >> l >> r;
        cout << query(r) - query(l - 1) << '\n';
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
