```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 50005;
const int MAX_CELLS = 200005; // 4 * n, n <= 50000

struct Cell {
    int x, y, d;
    bool operator<(const Cell& other) const {
        if (d != other.d) return d < other.d;
        if (x != other.x) return x < other.x;
        return y < other.y;
    }
};

int dist(int x, int y) {
    int add = (x % 3 == 2 && y % 3 == 2) ? 2 : 0;
    return x + y + add;
}

bool is_table(int x, int y) {
    int mx = x % 3;
    int my = y % 3;
    return (mx == 1 || mx == 2) && (my == 1 || my == 2);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int q;
    cin >> q;
    while (q--) {
        int n;
        cin >> n;
        vector<int> t(n);
        for (int i = 0; i < n; ++i) {
            cin >> t[i];
        }

        // Generate first 4*n table cells sorted by distance, then x, then y
        vector<Cell> cells;
        cells.reserve(MAX_CELLS);
        // We'll generate by distance up to some limit
        // Since max distance needed is around 3*sqrt(n) or so, but we can just generate until we have 4*n cells
        // Use BFS-like generation from (0,0) but only table cells
        // Actually we can iterate over possible distances
        // Max distance for 4*n cells is bounded by O(sqrt(n)*something) but we can just generate by increasing distance
        // Simpler: generate all table cells with x+y <= some bound, sort
        // Bound: we need 4*n cells. Distance roughly x+y, so max distance ~ 3*sqrt(4*n) ~ 600 for n=50000
        // Let's just generate up to x,y < 2000, that's enough
        for (int x = 0; x < 2000; ++x) {
            for (int y = 0; y < 2000; ++y) {
                if (is_table(x, y)) {
                    cells.push_back({x, y, dist(x, y)});
                }
            }
        }
        sort(cells.begin(), cells.end());
        if ((int)cells.size() > 4 * n) cells.resize(4 * n);

        int total_cells = cells.size();
        // Map cell coordinates to index in sorted order
        map<pair<int,int>, int> cell_to_idx;
        for (int i = 0; i < total_cells; ++i) {
            cell_to_idx[{cells[i].x, cells[i].y}] = i;
        }

        // occupied[i] = true if cell i is occupied
        vector<bool> occupied(total_cells, false);
        // For type 0: need completely unoccupied table
        // A table consists of 4 cells: (3a+1,3b+1), (3a+1,3b+2), (3a+2,3b+1), (3a+2,3b+2)
        // We can map each table cell to its table id (a,b)
        // table_id = (x/3, y/3) for table cells? Actually x = 3a+1 or 3a+2, so a = (x-1)/3
        auto get_table = [](int x, int y) -> pair<int,int> {
            return {(x-1)/3, (y-1)/3};
        };
        // table_occupied_count[{a,b}] = number of occupied cells in that table
        map<pair<int,int>, int> table_occ;

        // For each guest
        for (int i = 0; i < n; ++i) {
            int type = t[i];
            int chosen_idx = -1;
            // Find first suitable cell
            for (int j = 0; j < total_cells; ++j) {
                if (occupied[j]) continue;
                if (type == 1) {
                    chosen_idx = j;
                    break;
                } else {
                    // type 0: table must be completely unoccupied
                    auto [a, b] = get_table(cells[j].x, cells[j].y);
                    if (table_occ.find({a,b}) == table_occ.end() || table_occ[{a,b}] == 0) {
                        chosen_idx = j;
                        break;
                    }
                }
            }
            // Mark chosen cell as occupied
            occupied[chosen_idx] = true;
            auto [a, b] = get_table(cells[chosen_idx].x, cells[chosen_idx].y);
            table_occ[{a,b}]++;

            cout << cells[chosen_idx].x << " " << cells[chosen_idx].y << "\n";
        }
    }
    return 0;
}
```
