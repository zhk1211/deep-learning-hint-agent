```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n), b(n);
        for (int i = 0; i < n; ++i) cin >> a[i];
        for (int i = 0; i < n; ++i) cin >> b[i];
        int m;
        cin >> m;
        vector<int> x(m);
        for (int i = 0; i < m; ++i) cin >> x[i];

        bool possible = true;
        for (int i = 0; i < n; ++i) {
            if (a[i] < b[i]) {
                possible = false;
                break;
            }
        }
        if (!possible) {
            cout << "NO\n";
            continue;
        }

        map<int, int> needed;
        vector<int> stk;
        for (int i = 0; i < n; ++i) {
            while (!stk.empty() && stk.back() < b[i]) {
                stk.pop_back();
            }
            if (a[i] != b[i]) {
                if (stk.empty() || stk.back() != b[i]) {
                    needed[b[i]]++;
                    stk.push_back(b[i]);
                }
            }
        }

        map<int, int> have;
        for (int v : x) have[v]++;

        for (auto &p : needed) {
            if (have[p.first] < p.second) {
                possible = false;
                break;
            }
        }

        cout << (possible ? "YES" : "NO") << '\n';
    }
    return 0;
}
```
