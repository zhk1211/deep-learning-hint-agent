#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> p(n);
        for (int i = 0; i < n; ++i) {
            cin >> p[i];
        }
        
        // Find longest prefix that matches 1,2,3...
        int a = 0;
        for (int i = 0; i < n; ++i) {
            if (p[i] == i + 1) {
                ++a;
            } else {
                break;
            }
        }
        
        // Find longest suffix that matches n, n-1, ...
        int b = 0;
        for (int i = n - 1; i >= 0; --i) {
            if (p[i] == n - (n - 1 - i)) {
                ++b;
            } else {
                break;
            }
        }
        
        // c is the number of elements that are neither in the correct ascending prefix
        // nor in the correct descending suffix.
        int c = n - a - b;
        if (c < 0) c = 0; // just in case of overlap, though shouldn't happen with given constraints
        
        // Determine winner based on analysis from hints
        // First wins if a + c <= b (they can color all needed elements before Second wins)
        // Second wins if b + c < a (Second can color all needed elements before First wins)
        // Otherwise Tie
        if (a + c <= b) {
            cout << "First\n";
        } else if (b + c < a) {
            cout << "Second\n";
        } else {
            cout << "Tie\n";
        }
    }
    
    return 0;
}
