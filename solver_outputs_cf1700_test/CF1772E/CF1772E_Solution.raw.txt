#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> p(n);
        for (int i = 0; i < n; ++i) {
            cin >> p[i];
        }
        
        int a = 0, b = 0, c = 0;
        for (int i = 0; i < n; ++i) {
            bool need_first = (p[i] != i + 1);
            bool need_second = (p[i] != n - i);
            if (need_first && need_second) {
                ++c;
            } else if (need_first) {
                ++a;
            } else if (need_second) {
                ++b;
            }
        }
        
        if (a + c <= b) {
            cout << "First\n";
        } else if (b + c < a) {
            cout << "Second\n";
        } else {
            cout << "Tie\n";
        }
    }
    
    return 0;
}
