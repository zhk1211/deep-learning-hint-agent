```cpp
#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n;
    cin >> n;
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < n - 1; ++i) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    // Check if tree is a star: one center connected to all leaves
    int center = -1;
    for (int i = 1; i <= n; ++i) {
        if (adj[i].size() == n - 1) {
            center = i;
            break;
        }
    }
    
    if (center == -1) {
        cout << "NO\n";
        return;
    }
    
    cout << "YES\n";
    // Direct all edges from center to leaves
    for (int leaf : adj[center]) {
        cout << center << " " << leaf << "\n";
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
```
