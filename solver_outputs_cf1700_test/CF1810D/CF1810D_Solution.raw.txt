#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        int q;
        cin >> q;
        ll L_min = 1, L_max = 4e18; // large enough
        bool known = false;
        while (q--) {
            int type;
            cin >> type;
            if (type == 1) {
                ll a, b, n;
                cin >> a >> b >> n;
                ll l, r;
                if (n == 1) {
                    l = 1;
                    r = a;
                } else {
                    l = (n - 2) * (a - b) + a + 1;
                    r = (n - 1) * (a - b) + a;
                }
                if (!known) {
                    L_min = l;
                    L_max = r;
                    known = true;
                    cout << 1 << ' ';
                } else {
                    ll new_l = max(L_min, l);
                    ll new_r = min(L_max, r);
                    if (new_l <= new_r) {
                        L_min = new_l;
                        L_max = new_r;
                        cout << 1 << ' ';
                    } else {
                        cout << 0 << ' ';
                    }
                }
            } else {
                ll a, b;
                cin >> a >> b;
                if (!known) {
                    cout << -1 << ' ';
                } else {
                    auto days = [&](ll h) -> ll {
                        if (h <= a) return 1;
                        // ceil((h - a) / (a - b)) + 1
                        ll diff = h - a;
                        ll step = a - b;
                        return (diff + step - 1) / step + 1;
                    };
                    ll d1 = days(L_min);
                    ll d2 = days(L_max);
                    if (d1 == d2) cout << d1 << ' ';
                    else cout << -1 << ' ';
                }
            }
        }
        cout << '\n';
    }
    return 0;
}
