```cpp
#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, a, b;
    cin >> n >> a >> b;
    --a; --b;
    vector<vector<int>> g(n);
    for (int i = 0; i < n - 1; ++i) {
        int u, v;
        cin >> u >> v;
        --u; --v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    
    auto bfs = [&](int s) {
        vector<int> dist(n, -1);
        queue<int> q;
        dist[s] = 0;
        q.push(s);
        while (!q.empty()) {
            int u = q.front(); q.pop();
            for (int v : g[u]) {
                if (dist[v] == -1) {
                    dist[v] = dist[u] + 1;
                    q.push(v);
                }
            }
        }
        return dist;
    };
    
    vector<int> da = bfs(a), db = bfs(b);
    
    // find the vertex r that minimizes max(da[i], db[i]) but with da[i] <= db[i]?
    // Actually we need the first vertex painted blue.
    // P_A and P_B move towards each other. The meeting point is where they first can paint blue.
    // The first blue vertex r is the one that minimizes the time when P_A arrives and P_B arrives later.
    // The time when P_A reaches v is da[v], P_B reaches v is db[v].
    // The step when v becomes blue is max(da[v], db[v])? Wait: P_A moves first, then P_B.
    // If da[v] <= db[v], P_A arrives at step da[v], paints red. Then P_B arrives at step db[v], paints blue.
    // So blue time = db[v]. If da[v] > db[v], P_A arrives later, but P_B arrives earlier and v is white, so P_B cannot paint blue.
    // Actually P_B can only paint blue if v is red when P_B arrives. So we need da[v] <= db[v].
    // The first blue vertex is the one that minimizes db[v] subject to da[v] <= db[v].
    // But also they can meet halfway: they move simultaneously. The meeting point is on the path between a and b.
    // The first blue vertex is the one on the path where da[v] <= db[v] and db[v] is minimized.
    // Actually the official solution says: the first blue vertex r is the one that minimizes the meeting time.
    // Let's find r by moving along the path from a to b.
    // We can find the midpoint: the vertex where da[v] and db[v] are as close as possible with da[v] <= db[v].
    // The time to first blue is db[r].
    
    int r = a;
    int best = 1e9;
    for (int i = 0; i < n; ++i) {
        if (da[i] <= db[i]) {
            if (db[i] < best) {
                best = db[i];
                r = i;
            }
        }
    }
    
    // Now compute the maximum distance from r
    auto dr = bfs(r);
    int maxd = *max_element(dr.begin(), dr.end());
    
    // Answer = 2*(n-1) - maxd + (time to first blue)
    // But careful: the steps before first blue are already counted in the 2*(n-1) - maxd?
    // Official solution: answer = 2*(n-1) - d + (time to first blue).
    // Actually the formula: total steps = (steps to first blue) + (steps to cover rest).
    // After first blue, P_A can follow P_B's path. The remaining steps = 2*(n-1) - maxd.
    // So total = db[r] + 2*(n-1) - maxd.
    
    long long ans = (long long)db[r] + 2LL * (n - 1) - maxd;
    cout << ans << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
```
