#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        long long d, k;
        cin >> d >> k;
        
        // Find maximum z such that (k*z, k*z) is within distance d
        // i.e., 2 * (k*z)^2 <= d^2
        // (k*z)^2 <= d^2 / 2
        // k*z <= d / sqrt(2)
        long long z = (long long)(d / (sqrt(2) * k));
        
        // Check if we can move one more step in x direction from (k*z, k*z)
        // i.e., check if (k*(z+1), k*z) is within distance d
        long long x = k * (z + 1);
        long long y = k * z;
        if (x * x + y * y <= d * d) {
            cout << "Ashish\n";
        } else {
            cout << "Utkarsh\n";
        }
    }
    
    return 0;
}
