#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<string> grid(n);
        for (int i = 0; i < n; ++i) {
            cin >> grid[i];
        }

        vector<vector<int>> mark(n, vector<int>(m, 0));
        int comp = 0;
        bool ok = true;

        // First pass: check neighbor counts and mark elbows
        for (int i = 0; i < n && ok; ++i) {
            for (int j = 0; j < m && ok; ++j) {
                if (grid[i][j] != '*') continue;
                int cnt = 0;
                if (i > 0 && grid[i-1][j] == '*') cnt++;
                if (i+1 < n && grid[i+1][j] == '*') cnt++;
                if (j > 0 && grid[i][j-1] == '*') cnt++;
                if (j+1 < m && grid[i][j+1] == '*') cnt++;

                if (cnt > 2) {
                    ok = false;
                    break;
                }
                if (cnt == 2) {
                    // Check if two neighbors are opposite (straight line)
                    bool up = (i > 0 && grid[i-1][j] == '*');
                    bool down = (i+1 < n && grid[i+1][j] == '*');
                    bool left = (j > 0 && grid[i][j-1] == '*');
                    bool right = (j+1 < m && grid[i][j+1] == '*');
                    if ((up && down) || (left && right)) {
                        ok = false;
                        break;
                    }
                    // It's an elbow, mark the three cells
                    comp++;
                    mark[i][j] = comp;
                    if (up) mark[i-1][j] = comp;
                    if (down) mark[i+1][j] = comp;
                    if (left) mark[i][j-1] = comp;
                    if (right) mark[i][j+1] = comp;
                }
            }
        }

        if (!ok) {
            cout << "NO\n";
            continue;
        }

        // Check if all shaded cells are marked
        for (int i = 0; i < n && ok; ++i) {
            for (int j = 0; j < m && ok; ++j) {
                if (grid[i][j] == '*' && mark[i][j] == 0) {
                    ok = false;
                }
            }
        }

        if (!ok) {
            cout << "NO\n";
            continue;
        }

        // Check diagonal adjacency between different components
        for (int i = 0; i < n && ok; ++i) {
            for (int j = 0; j < m && ok; ++j) {
                if (mark[i][j] == 0) continue;
                int cur = mark[i][j];
                for (int di = -1; di <= 1; di += 2) {
                    for (int dj = -1; dj <= 1; dj += 2) {
                        int ni = i + di;
                        int nj = j + dj;
                        if (ni >= 0 && ni < n && nj >= 0 && nj < m) {
                            if (mark[ni][nj] != 0 && mark[ni][nj] != cur) {
                                ok = false;
                            }
                        }
                    }
                }
            }
        }

        cout << (ok ? "YES" : "NO") << "\n";
    }
    return 0;
}
