```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    string s;
    cin >> n >> s;

    // Find first '1'
    int first_one = -1;
    for (int i = 0; i < n; ++i) {
        if (s[i] == '1') {
            first_one = i;
            break;
        }
    }

    if (first_one == -1) {
        cout << "0\n";
        return 0;
    }

    // The first substring will be the whole string from first_one to end
    string best = s.substr(first_one);
    int m = best.size();

    // Find the second '1' position (first zero after first_one)
    int first_zero = -1;
    for (int i = 0; i < m; ++i) {
        if (best[i] == '0') {
            first_zero = i;
            break;
        }
    }

    if (first_zero == -1) {
        // All ones
        cout << best << "\n";
        return 0;
    }

    // We need to find a substring that starts at some position in [0, first_zero]
    // and has length m, to maximize OR with best.
    // Since best has a zero at first_zero, we want to find a substring that has '1' there.
    // We only need to consider substrings starting at positions 0..first_zero.
    // Among those, we can compare them efficiently.
    string ans = best;
    for (int start = 1; start <= first_zero; ++start) {
        // candidate substring of length m starting at start in original s
        // but we only have best which starts at first_one, so candidate starts at start relative to best
        string cand = best;
        for (int i = first_zero; i < m; ++i) {
            if (best[i] == '1') continue;
            // we need to check if s[first_one + start + i] == '1'
            if (s[first_one + start + i] == '1') {
                cand[i] = '1';
            }
        }
        if (cand > ans) ans = cand;
    }

    cout << ans << "\n";
    return 0;
}
```
