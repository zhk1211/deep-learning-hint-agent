```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }
        
        vector<int> pos(n + 1);
        for (int i = 0; i < n; ++i) {
            pos[a[i]] = i;
        }
        
        int l = pos[n], r = pos[n];
        int ans = 0;
        for (int v = n - 1; v >= 1; --v) {
            int p = pos[v];
            if (p < l) {
                l = p;
            } else if (p > r) {
                r = p;
            } else {
                ans = v;
                break;
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
```
