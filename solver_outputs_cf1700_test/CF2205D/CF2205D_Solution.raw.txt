#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }
        
        vector<int> left(n, -1), right(n, -1);
        vector<int> parent(n, -1);
        stack<int> st;
        for (int i = 0; i < n; ++i) {
            int last = -1;
            while (!st.empty() && a[st.top()] < a[i]) {
                last = st.top();
                st.pop();
            }
            if (!st.empty()) {
                parent[i] = st.top();
                right[st.top()] = i;
            }
            if (last != -1) {
                parent[last] = i;
                left[i] = last;
            }
            st.push(i);
        }
        
        int root = -1;
        for (int i = 0; i < n; ++i) {
            if (parent[i] == -1) {
                root = i;
                break;
            }
        }
        
        vector<int> depth(n, 0);
        int max_depth = 0;
        function<void(int, int)> dfs = [&](int u, int d) {
            depth[u] = d;
            max_depth = max(max_depth, d);
            if (left[u] != -1) dfs(left[u], d + 1);
            if (right[u] != -1) dfs(right[u], d + 1);
        };
        dfs(root, 1);
        
        cout << n - max_depth << '\n';
    }
    return 0;
}
