#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }
        
        vector<int> left(n, -1), right(n, -1);
        vector<int> st;
        for (int i = 0; i < n; ++i) {
            int last = -1;
            while (!st.empty() && a[st.back()] < a[i]) {
                last = st.back();
                st.pop_back();
            }
            if (!st.empty()) {
                right[st.back()] = i;
            }
            if (last != -1) {
                left[i] = last;
            }
            st.push_back(i);
        }
        
        int root = st.empty() ? -1 : st[0];
        vector<int> depth(n, 0);
        int max_depth = 0;
        
        function<void(int, int)> dfs = [&](int u, int d) {
            if (u == -1) return;
            depth[u] = d;
            max_depth = max(max_depth, d);
            dfs(left[u], d + 1);
            dfs(right[u], d + 1);
        };
        
        dfs(root, 1);
        
        cout << n - max_depth << '\n';
    }
    
    return 0;
}
