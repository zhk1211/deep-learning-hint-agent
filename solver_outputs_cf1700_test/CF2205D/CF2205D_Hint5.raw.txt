#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }

        vector<int> left(n, -1), right(n, -1);
        vector<int> stk;
        vector<int> parent(n, -1);
        int root = -1;

        for (int i = 0; i < n; ++i) {
            int last = -1;
            while (!stk.empty() && a[stk.back()] < a[i]) {
                last = stk.back();
                stk.pop_back();
            }
            if (!stk.empty()) {
                parent[i] = stk.back();
                right[stk.back()] = i;
            } else {
                root = i;
            }
            if (last != -1) {
                parent[last] = i;
                left[i] = last;
            }
            stk.push_back(i);
        }

        vector<int> depth(n, 0);
        function<void(int)> dfs = [&](int u) {
            if (left[u] != -1) {
                depth[left[u]] = depth[u] + 1;
                dfs(left[u]);
            }
            if (right[u] != -1) {
                depth[right[u]] = depth[u] + 1;
                dfs(right[u]);
            }
        };
        dfs(root);

        int max_depth = *max_element(depth.begin(), depth.end());
        cout << n - (max_depth + 1) << '\n';
    }
    return 0;
}
