```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n, q;
        cin >> n >> q;
        vector<int> p(n + 1);
        vector<int> pos(n + 1);
        for (int i = 1; i <= n; ++i) {
            cin >> p[i];
            pos[p[i]] = i;
        }
        string s;
        cin >> s;
        s = " " + s; // 1-indexed

        vector<int> d(n + 2, 0);
        for (int x = 1; x <= n; ++x) {
            int l = min(x, pos[x]);
            int r = max(x, pos[x]);
            if (l < r) {
                d[l]++;
                d[r]--;
            }
        }
        for (int i = 1; i <= n; ++i) {
            d[i] += d[i - 1];
        }

        auto is_bad = [&](int i) -> bool {
            if (i < 1 || i >= n) return false;
            return s[i] == 'L' && s[i + 1] == 'R';
        };

        set<int> bad;
        for (int i = 1; i < n; ++i) {
            if (d[i] > 0 && is_bad(i)) {
                bad.insert(i);
            }
        }

        while (q--) {
            int idx;
            cin >> idx;
            // flip s[idx]
            s[idx] = (s[idx] == 'L' ? 'R' : 'L');

            // indices that could change: idx-1 and idx
            for (int cand : {idx - 1, idx}) {
                if (cand >= 1 && cand < n) {
                    if (d[cand] > 0 && is_bad(cand)) {
                        bad.insert(cand);
                    } else {
                        bad.erase(cand);
                    }
                }
            }

            cout << (bad.empty() ? "YES" : "NO") << '\n';
        }
    }
    return 0;
}
```
