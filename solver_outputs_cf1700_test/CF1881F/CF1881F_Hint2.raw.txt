#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, k;
    cin >> n >> k;
    vector<int> marked(k);
    for (int i = 0; i < k; ++i) {
        cin >> marked[i];
        --marked[i];
    }
    vector<vector<int>> adj(n);
    for (int i = 0; i < n - 1; ++i) {
        int u, v;
        cin >> u >> v;
        --u; --v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    if (k == 1) {
        cout << 0 << '\n';
        return;
    }
    vector<int> dist(n, -1);
    queue<int> q;
    q.push(marked[0]);
    dist[marked[0]] = 0;
    int far = marked[0];
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : adj[u]) {
            if (dist[v] == -1) {
                dist[v] = dist[u] + 1;
                q.push(v);
                if (find(marked.begin(), marked.end(), v) != marked.end() && dist[v] > dist[far]) {
                    far = v;
                }
            }
        }
    }
    int a = far;
    fill(dist.begin(), dist.end(), -1);
    q.push(a);
    dist[a] = 0;
    far = a;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : adj[u]) {
            if (dist[v] == -1) {
                dist[v] = dist[u] + 1;
                q.push(v);
                if (find(marked.begin(), marked.end(), v) != marked.end() && dist[v] > dist[far]) {
                    far = v;
                }
            }
        }
    }
    int b = far;
    vector<int> distA(n, -1), distB(n, -1);
    q.push(a);
    distA[a] = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : adj[u]) {
            if (distA[v] == -1) {
                distA[v] = distA[u] + 1;
                q.push(v);
            }
        }
    }
    q.push(b);
    distB[b] = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : adj[u]) {
            if (distB[v] == -1) {
                distB[v] = distB[u] + 1;
                q.push(v);
            }
        }
    }
    int ans = INT_MAX;
    for (int i = 0; i < n; ++i) {
        ans = min(ans, max(distA[i], distB[i]));
    }
    cout << ans << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
