```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int a, b, c;
        cin >> a >> b >> c;
        
        if (c != a + 1) {
            cout << -1 << '\n';
            continue;
        }
        
        int height = 0;
        int cur = 1; // nodes in current level that can have children
        int nxt = 0; // nodes in next level
        
        while (a > 0 || b > 0) {
            if (cur == 0) {
                cur = nxt;
                nxt = 0;
                height++;
            }
            if (a > 0) {
                a--;
                cur--;
                nxt += 2;
            } else {
                b--;
                cur--;
                nxt += 1;
            }
        }
        
        cout << height << '\n';
    }
    
    return 0;
}
```
