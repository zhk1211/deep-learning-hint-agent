#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, m;
    cin >> n >> m;
    vector<string> grid(n);
    for (int i = 0; i < n; ++i) {
        cin >> grid[i];
    }

    vector<vector<int>> comp_id(n, vector<int>(m, -1));
    vector<int> comp_size;
    int comp_cnt = 0;

    const int dx[] = {1, -1, 0, 0};
    const int dy[] = {0, 0, 1, -1};

    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            if (grid[i][j] == '#' && comp_id[i][j] == -1) {
                int sz = 0;
                queue<pair<int,int>> q;
                q.push({i, j});
                comp_id[i][j] = comp_cnt;
                while (!q.empty()) {
                    auto [x, y] = q.front(); q.pop();
                    sz++;
                    for (int d = 0; d < 4; ++d) {
                        int nx = x + dx[d];
                        int ny = y + dy[d];
                        if (nx >= 0 && nx < n && ny >= 0 && ny < m &&
                            grid[nx][ny] == '#' && comp_id[nx][ny] == -1) {
                            comp_id[nx][ny] = comp_cnt;
                            q.push({nx, ny});
                        }
                    }
                }
                comp_size.push_back(sz);
                comp_cnt++;
            }
        }
    }

    int ans = 0;

    // Rows
    for (int r = 0; r < n; ++r) {
        int dots = 0;
        for (int j = 0; j < m; ++j) {
            if (grid[r][j] == '.') dots++;
        }
        set<int> comps;
        for (int dr = -1; dr <= 1; ++dr) {
            int nr = r + dr;
            if (nr < 0 || nr >= n) continue;
            for (int j = 0; j < m; ++j) {
                if (grid[nr][j] == '#') {
                    comps.insert(comp_id[nr][j]);
                }
            }
        }
        int total = dots;
        for (int cid : comps) {
            total += comp_size[cid];
        }
        ans = max(ans, total);
    }

    // Columns
    for (int c = 0; c < m; ++c) {
        int dots = 0;
        for (int i = 0; i < n; ++i) {
            if (grid[i][c] == '.') dots++;
        }
        set<int> comps;
        for (int dc = -1; dc <= 1; ++dc) {
            int nc = c + dc;
            if (nc < 0 || nc >= m) continue;
            for (int i = 0; i < n; ++i) {
                if (grid[i][nc] == '#') {
                    comps.insert(comp_id[i][nc]);
                }
            }
        }
        int total = dots;
        for (int cid : comps) {
            total += comp_size[cid];
        }
        ans = max(ans, total);
    }

    cout << ans << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
