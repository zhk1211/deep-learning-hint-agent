```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<ll> a(n), b(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i] >> b[i];
        }
        
        int ans = 1;
        ll g = 0, l = 1;
        for (int i = 0; i < n; ++i) {
            ll val = a[i] * b[i];
            g = gcd(g, val);
            l = l / gcd(l, b[i]) * b[i];
            if (g % l != 0) {
                ++ans;
                g = val;
                l = b[i];
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
```
