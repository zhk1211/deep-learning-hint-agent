#include <bits/stdc++.h>
using namespace std;

const int MOD = 998244353;

int modpow(int a, int e) {
    int res = 1;
    while (e) {
        if (e & 1) res = 1LL * res * a % MOD;
        a = 1LL * a * a % MOD;
        e >>= 1;
    }
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        int n, q;
        cin >> n >> q;
        vector<int> a(n), b(n);
        for (int i = 0; i < n; ++i) cin >> a[i];
        for (int i = 0; i < n; ++i) cin >> b[i];
        vector<int> sa = a, sb = b;
        sort(sa.begin(), sa.end());
        sort(sb.begin(), sb.end());
        long long cur = 1;
        for (int i = 0; i < n; ++i) {
            cur = cur * min(sa[i], sb[i]) % MOD;
        }
        cout << cur << ' ';
        while (q--) {
            int o, x;
            cin >> o >> x;
            --x;
            if (o == 1) {
                int val = a[x];
                a[x]++;
                int pos = upper_bound(sa.begin(), sa.end(), val) - sa.begin() - 1;
                cur = cur * modpow(min(sa[pos], sb[pos]), MOD - 2) % MOD;
                sa[pos]++;
                cur = cur * min(sa[pos], sb[pos]) % MOD;
            } else {
                int val = b[x];
                b[x]++;
                int pos = upper_bound(sb.begin(), sb.end(), val) - sb.begin() - 1;
                cur = cur * modpow(min(sa[pos], sb[pos]), MOD - 2) % MOD;
                sb[pos]++;
                cur = cur * min(sa[pos], sb[pos]) % MOD;
            }
            cout << cur << ' ';
        }
        cout << '\n';
    }
    return 0;
}
