```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        int n = (int)s.size();
        int ans = 0;
        
        for (int d = 1; d <= n / 2; ++d) {
            int cnt = 0;
            for (int i = 0; i < n - d; ++i) {
                if (s[i] == s[i + d] || s[i] == '?' || s[i + d] == '?')
                    ++cnt;
                if (i - d >= 0) {
                    if (s[i - d] == s[i] || s[i - d] == '?' || s[i] == '?')
                        --cnt;
                }
                if (i - d >= -1 && cnt == d)
                    ans = 2 * d;
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
```
