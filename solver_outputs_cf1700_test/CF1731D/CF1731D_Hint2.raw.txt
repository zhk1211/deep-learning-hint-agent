#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<vector<int>> a(n, vector<int>(m));
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                cin >> a[i][j];
            }
        }
        
        auto check = [&](int l) -> bool {
            if (l > n || l > m) return false;
            vector<vector<int>> pref(n + 1, vector<int>(m + 1, 0));
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    pref[i + 1][j + 1] = (a[i][j] >= l) + pref[i][j + 1] + pref[i + 1][j] - pref[i][j];
                }
            }
            for (int i = l; i <= n; i++) {
                for (int j = l; j <= m; j++) {
                    if (pref[i][j] - pref[i - l][j] - pref[i][j - l] + pref[i - l][j - l] == l * l) {
                        return true;
                    }
                }
            }
            return false;
        };
        
        int lo = 1, hi = min(n, m), ans = 1;
        while (lo <= hi) {
            int mid = (lo + hi) / 2;
            if (check(mid)) {
                ans = mid;
                lo = mid + 1;
            } else {
                hi = mid - 1;
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
