```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAX_A = 1000000;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        vector<int> freq(MAX_A + 1, 0);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
            freq[a[i]]++;
        }

        long long ans = 0;
        for (int x = 1; x <= MAX_A; ++x) {
            if (freq[x] == 0) continue;
            // b = 1 case: (x, x, x)
            if (freq[x] >= 3) {
                ans += 1LL * freq[x] * (freq[x] - 1) * (freq[x] - 2);
            }
            // b > 1 case
            for (int b = 2; 1LL * x * b * b <= MAX_A; ++b) {
                int y = x * b;
                int z = y * b;
                if (z > MAX_A) break;
                if (freq[y] > 0 && freq[z] > 0) {
                    ans += 1LL * freq[x] * freq[y] * freq[z];
                }
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
```
