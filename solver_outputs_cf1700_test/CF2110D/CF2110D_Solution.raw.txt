#include <bits/stdc++.h>
using namespace std;

const long long INF = 1e18;

void solve() {
    int n, m;
    cin >> n >> m;
    vector<long long> b(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> b[i];
    }
    vector<vector<pair<int, long long>>> g(n + 1);
    for (int i = 0; i < m; ++i) {
        int u, v;
        long long w;
        cin >> u >> v >> w;
        g[u].emplace_back(v, w);
    }

    // Check if node n is reachable at all (ignoring battery constraints)
    vector<bool> reachable(n + 1, false);
    reachable[1] = true;
    for (int v = 1; v <= n; ++v) {
        if (!reachable[v]) continue;
        for (auto &[to, w] : g[v]) {
            reachable[to] = true;
        }
    }
    if (!reachable[n]) {
        cout << -1 << '\n';
        return;
    }

    long long low = 0, high = 0;
    for (int i = 1; i <= n; ++i) high += b[i];
    high = min(high, (long long)1e18);
    long long ans = -1;

    while (low <= high) {
        long long mid = (low + high) / 2;
        vector<long long> dp(n + 1, -INF);
        dp[1] = 0;
        for (int v = 1; v <= n; ++v) {
            if (dp[v] < 0) continue;
            dp[v] += b[v];
            if (dp[v] > mid) dp[v] = mid;
            for (auto &[to, w] : g[v]) {
                if (dp[v] >= w) {
                    if (dp[v] > dp[to]) dp[to] = dp[v];
                }
            }
        }
        if (dp[n] >= 0) {
            ans = mid;
            high = mid - 1;
        } else {
            low = mid + 1;
        }
    }
    cout << ans << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
