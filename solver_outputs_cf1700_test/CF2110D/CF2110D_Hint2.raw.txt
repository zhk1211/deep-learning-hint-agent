```cpp
#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

const ll INF = 1e18;

int n, m;
vector<ll> b;
vector<vector<pair<int, ll>>> g;

bool can(ll start) {
    vector<ll> dp(n + 1, -INF);
    dp[1] = start;
    for (int v = 1; v <= n; ++v) {
        if (dp[v] < 0) continue;
        dp[v] = min(dp[v] + b[v], INF);
        for (auto &e : g[v]) {
            int to = e.first;
            ll w = e.second;
            if (dp[v] >= w) {
                dp[to] = max(dp[to], dp[v]);
            }
        }
    }
    return dp[n] >= 0;
}

void solve() {
    cin >> n >> m;
    b.resize(n + 1);
    for (int i = 1; i <= n; ++i) cin >> b[i];
    g.assign(n + 1, {});
    for (int i = 0; i < m; ++i) {
        int u, v;
        ll w;
        cin >> u >> v >> w;
        g[u].push_back({v, w});
    }
    
    // check reachability first
    vector<bool> vis(n + 1, false);
    queue<int> q;
    q.push(1);
    vis[1] = true;
    while (!q.empty()) {
        int v = q.front(); q.pop();
        for (auto &e : g[v]) {
            int to = e.first;
            if (!vis[to]) {
                vis[to] = true;
                q.push(to);
            }
        }
    }
    if (!vis[n]) {
        cout << "-1\n";
        return;
    }
    
    ll lo = 0, hi = 1e18, ans = -1;
    while (lo <= hi) {
        ll mid = lo + (hi - lo) / 2;
        if (can(mid)) {
            ans = mid;
            hi = mid - 1;
        } else {
            lo = mid + 1;
        }
    }
    cout << ans << "\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
```
