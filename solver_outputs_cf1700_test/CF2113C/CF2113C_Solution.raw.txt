#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, m, k;
    cin >> n >> m >> k;
    vector<string> grid(n);
    for (int i = 0; i < n; ++i) {
        cin >> grid[i];
    }

    vector<vector<int>> pref(n + 1, vector<int>(m + 1, 0));
    int total_gold = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            int val = (grid[i][j] == 'g') ? 1 : 0;
            total_gold += val;
            pref[i + 1][j + 1] = pref[i][j + 1] + pref[i + 1][j] - pref[i][j] + val;
        }
    }

    auto get_sum = [&](int r1, int c1, int r2, int c2) -> int {
        if (r1 > r2 || c1 > c2) return 0;
        r1 = max(0, r1);
        c1 = max(0, c1);
        r2 = min(n - 1, r2);
        c2 = min(m - 1, c2);
        if (r1 > r2 || c1 > c2) return 0;
        return pref[r2 + 1][c2 + 1] - pref[r1][c2 + 1] - pref[r2 + 1][c1] + pref[r1][c1];
    };

    int min_loss = total_gold;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            if (grid[i][j] != '.') continue;
            int r1 = i - k;
            int c1 = j - k;
            int r2 = i + k;
            int c2 = j + k;
            int inside = get_sum(r1 + 1, c1 + 1, r2 - 1, c2 - 1);
            min_loss = min(min_loss, inside);
        }
    }

    cout << total_gold - min_loss << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
