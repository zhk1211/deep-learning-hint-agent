#include <bits/stdc++.h>
using namespace std;

const int MOD = 1e9 + 7;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    string s;
    cin >> s;
    int n = (int)s.size();
    
    vector<long long> pow10(n + 1);
    pow10[0] = 1;
    for (int i = 1; i <= n; i++) {
        pow10[i] = (pow10[i-1] * 10) % MOD;
    }
    
    vector<long long> pref(n + 1, 0);
    for (int i = 0; i < n; i++) {
        pref[i+1] = (pref[i] * 10 + (s[i] - '0')) % MOD;
    }
    
    long long total_sum = 0;
    long long suffix_sum = 0;
    long long suffix_pow = 1;
    
    for (int i = n - 1; i >= 0; i--) {
        int digit = s[i] - '0';
        long long left_val = pref[i];
        long long left_len = i;
        
        long long contrib = (left_val * pow10[n - 1 - i]) % MOD;
        contrib = (contrib * (i + 1)) % MOD;
        
        long long right_contrib = (digit * suffix_sum) % MOD;
        
        total_sum = (total_sum + contrib + right_contrib) % MOD;
        
        suffix_sum = (suffix_sum + suffix_pow * (i + 1)) % MOD;
        suffix_pow = (suffix_pow * 10) % MOD;
    }
    
    cout << total_sum % MOD << "\n";
    return 0;
}
