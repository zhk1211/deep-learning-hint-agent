```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }

        // Maximum XOR value is less than 2*n
        int max_xor = 1;
        while (max_xor <= 2 * n) {
            max_xor <<= 1;
        }
        max_xor -= 1;

        // Count subarrays with XOR equal to a perfect square
        long long odd_count = 0;
        vector<int> freq(max_xor + 1, 0);
        freq[0] = 1;
        int prefix = 0;

        for (int i = 0; i < n; ++i) {
            prefix ^= a[i];
            // Check all perfect squares <= 2*n
            for (int k = 0; k * k <= 2 * n; ++k) {
                int target = prefix ^ (k * k);
                if (target <= max_xor) {
                    odd_count += freq[target];
                }
            }
            freq[prefix]++;
        }

        long long total = (long long)n * (n + 1) / 2;
        long long even_count = total - odd_count;
        cout << even_count << '\n';
    }
    return 0;
}
```
