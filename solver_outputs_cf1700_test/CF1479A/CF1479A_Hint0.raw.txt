```cpp
#include <bits/stdc++.h>
using namespace std;

int query(int i) {
    cout << "? " << i << endl;
    int x;
    cin >> x;
    return x;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int n;
    cin >> n;
    
    if (n == 1) {
        cout << "! 1" << endl;
        return 0;
    }
    
    int a1 = query(1);
    int a2 = query(2);
    if (a1 < a2) {
        cout << "! 1" << endl;
        return 0;
    }
    
    int an = query(n);
    int an_1 = query(n - 1);
    if (an < an_1) {
        cout << "! " << n << endl;
        return 0;
    }
    
    int lo = 1, hi = n;
    while (lo <= hi) {
        int mid = (lo + hi) / 2;
        int amid = query(mid);
        int left = (mid > 1) ? query(mid - 1) : INT_MAX;
        int right = (mid < n) ? query(mid + 1) : INT_MAX;
        
        if (amid < left && amid < right) {
            cout << "! " << mid << endl;
            return 0;
        }
        
        if (amid > left) {
            hi = mid - 1;
        } else {
            lo = mid + 1;
        }
    }
    
    return 0;
}
```
