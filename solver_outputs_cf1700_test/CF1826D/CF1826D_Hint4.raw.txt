```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<long long> b(n);
        for (int i = 0; i < n; ++i) {
            cin >> b[i];
        }
        
        long long ans = LLONG_MIN;
        for (int i = 0; i < n; ++i) {
            long long left_best = LLONG_MIN;
            for (int j = 0; j < i; ++j) {
                left_best = max(left_best, b[j] + j);
            }
            long long right_best = LLONG_MIN;
            for (int j = i + 1; j < n; ++j) {
                right_best = max(right_best, b[j] - j);
            }
            if (left_best != LLONG_MIN && right_best != LLONG_MIN) {
                ans = max(ans, left_best + b[i] + right_best);
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
```
