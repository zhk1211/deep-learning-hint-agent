            }

        ans = INF;
        for (int top = 1; top <= n; ++top) {
            for (int bot = top + 4; bot <= n; ++bot) {
                int height = bot - top - 1; // number of inner rows
                vector<int> vert_cost(m + 1);
                vector<int> top_row(m + 1), bot_row(m + 1);

                    int ones = get_sum(top+1, c, bot-1, c);
                    vert_cost[c] = height - ones;
                    // top and bottom rows
                    top_row[c] = (grid[top-1][c-1] == '1') ? 1 : 0;
                    bot_row[c] = (grid[bot-1][c-1] == '1') ? 1 : 0;
                    // interior column ones
                    inside_col[c] = ones; // same as vertical ones, because interior is between top+1 and bot-1
                }
                // prefix sums
                vector<int> pref_top(m + 1, 0), pref_bot(m + 1, 0), pref_inside(m + 1, 0);
                for (int c = 1; c <= m; ++c) {
                    pref_top[c] = pref_top[c-1] + top_row[c];
                    pref_bot[c] = pref_bot[c-1] + bot_row[c];
                    pref_inside[c] = pref_inside[c-1] + inside_col[c];
                }

                vector<int> left_terms(m + 1);
                for (int left = 1; left <= m; ++left) {
                    left_terms[left] = vert_cost[left] - 2*left + pref_top[left] + pref_bot[left] - pref_inside[left];

                        ans = min(ans, min_left + right_terms);
                    }
                }
            }
        }
        cout << ans << '\n';
    }
    return 0;
