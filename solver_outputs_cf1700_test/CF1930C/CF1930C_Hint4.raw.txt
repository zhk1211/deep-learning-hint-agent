```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }

        vector<int> vals(n);
        for (int i = 0; i < n; ++i) {
            vals[i] = a[i] + (i + 1);
        }

        sort(vals.begin(), vals.end(), greater<int>());

        vector<int> ans;
        ans.push_back(vals[0]);
        for (int i = 1; i < n; ++i) {
            if (vals[i] >= ans.back()) {
                ans.push_back(ans.back() - 1);
            } else {
                ans.push_back(vals[i]);
            }
        }

        for (int i = 0; i < n; ++i) {
            cout << ans[i] << " \n"[i == n - 1];
        }
    }
    return 0;
}
```
