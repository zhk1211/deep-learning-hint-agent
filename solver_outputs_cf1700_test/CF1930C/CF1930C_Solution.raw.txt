```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }

        set<int> S;
        vector<int> b;
        for (int i = 0; i < n; ++i) {
            int val = a[i] + (i + 1);
            // find smallest v in [0, i] such that val - v not in S
            // we can maintain a pointer or use set operations
            // Since we have i+1 options and only i elements in S,
            // we can just try v from 0 upwards until we find one not in S.
            // To make it efficient, we can maintain the current candidate.
            // But simple loop is O(n^2) worst-case.
            // Better: maintain a set of used values and find the smallest missing
            // in the range [val - i, val].
            // We can use a variable to track the current candidate.
            // Actually, we can just maintain a set of used values and for each i,
            // we start from val and go down until we find a value not in S.
            // Since each value is inserted at most once, total operations O(n log n).
            int cur = val;
            while (S.count(cur)) {
                --cur;
            }
            S.insert(cur);
            b.push_back(cur);
        }

        sort(b.rbegin(), b.rend());
        for (int i = 0; i < n; ++i) {
            cout << b[i] << " \n"[i == n - 1];
        }
    }
    return 0;
}
```
