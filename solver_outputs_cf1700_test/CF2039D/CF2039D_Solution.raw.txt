```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    const int MAXN = 100000;
    vector<int> p(MAXN + 1, 0);
    for (int i = 2; i <= MAXN; ++i) {
        if (p[i] == 0) {
            for (int j = i; j <= MAXN; j += i) {
                int x = j;
                while (x % i == 0) {
                    p[j]++;
                    x /= i;
                }
            }
        }
    }

    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<int> s(m);
        for (int i = 0; i < m; ++i) {
            cin >> s[i];
        }

        int max_p = 0;
        for (int i = 1; i <= n; ++i) {
            max_p = max(max_p, p[i]);
        }
        if (m < max_p + 1) {
            cout << "-1\n";
            continue;
        }

        for (int i = 1; i <= n; ++i) {
            cout << s[m - 1 - p[i]] << (i == n ? '\n' : ' ');
        }
    }
    return 0;
}
```
