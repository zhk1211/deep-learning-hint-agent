#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n, m, i1, j1, i2, j2;
        string d;
        cin >> n >> m >> i1 >> j1 >> i2 >> j2 >> d;
        
        int di = (d[0] == 'D' ? 1 : -1);
        int dj = (d[1] == 'R' ? 1 : -1);
        
        int ci = i1, cj = j1;
        int bounces = 0;
        bool found = false;
        
        // We'll simulate up to 4*n*m steps to avoid infinite loops
        for (int step = 0; step < 4 * n * m; ++step) {
            if (ci == i2 && cj == j2) {
                cout << bounces << "\n";
                found = true;
                break;
            }
            
            // Check if next step hits a wall
            bool bounce = false;
            if (ci + di < 1 || ci + di > n) {
                di = -di;
                bounce = true;
            }
            if (cj + dj < 1 || cj + dj > m) {
                dj = -dj;
                bounce = true;
            }
            if (bounce) {
                ++bounces;
            }
            
            ci += di;
            cj += dj;
        }
        
        if (!found) {
            cout << -1 << "\n";
        }
    }
    
    return 0;
}
