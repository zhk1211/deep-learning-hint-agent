#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, m, i1, j1, i2, j2;
    string d;
    cin >> n >> m >> i1 >> j1 >> i2 >> j2 >> d;
    
    int di = (d[0] == 'D' ? 1 : -1);
    int dj = (d[1] == 'R' ? 1 : -1);
    
    int ci = i1, cj = j1;
    int bounces = 0;
    
    // To detect cycles, we track state: (i, j, di, dj)
    // Since n,m <= 25000, total states <= 4 * n * m, but sum n*m <= 5e4, so it's fine.
    // We'll use a set of tuple (i, j, di, dj).
    set<tuple<int,int,int,int>> visited;
    
    while (true) {
        if (ci == i2 && cj == j2) {
            cout << bounces << "\n";
            return;
        }
        
        auto state = make_tuple(ci, cj, di, dj);
        if (visited.count(state)) {
            cout << -1 << "\n";
            return;
        }
        visited.insert(state);
        
        // Check if next step hits wall
        int ni = ci + di;
        int nj = cj + dj;
        
        bool hit_i = false, hit_j = false;
        if (ni < 1 || ni > n) {
            hit_i = true;
            di = -di;
        }
        if (nj < 1 || nj > m) {
            hit_j = true;
            dj = -dj;
        }
        
        if (hit_i || hit_j) {
            bounces++;
        }
        
        // Move to next cell
        ci = ci + di;
        cj = cj + dj;
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
