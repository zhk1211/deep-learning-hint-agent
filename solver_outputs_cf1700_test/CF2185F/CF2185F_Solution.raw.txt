#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n, q;
        cin >> n >> q;
        int m = 1 << n;
        vector<int> a(m);
        for (int i = 0; i < m; ++i) {
            cin >> a[i];
        }
        
        vector<int> pref(m + 1, 0);
        for (int i = 0; i < m; ++i) {
            pref[i + 1] = pref[i] ^ a[i];
        }
        
        while (q--) {
            int b, c;
            cin >> b >> c;
            --b;
            
            int cur = b;
            int cur_skill = c;
            int cur_size = 1;
            int cur_xor = pref[cur] ^ pref[cur + 1];
            int ans = 0;
            
            for (int round = 0; round < n; ++round) {
                int block = 1 << round;
                int pos_in_block = cur % (2 * block);
                int opp_start, opp_end;
                if (pos_in_block < block) {
                    opp_start = cur + (block - pos_in_block);
                    opp_end = opp_start + block - 1;
                } else {
                    opp_start = cur - (pos_in_block - block + 1);
                    opp_end = opp_start + block - 1;
                }
                
                int opp_xor = pref[opp_end + 1] ^ pref[opp_start];
                int opp_skill = opp_xor ^ a[opp_start];
                
                if (cur_skill > opp_skill || (cur_skill == opp_skill && cur < opp_start)) {
                    cur_skill ^= opp_xor;
                    cur_size += block;
                    cur_xor ^= opp_xor;
                    if (pos_in_block >= block) {
                        ans += block;
                    }
                } else {
                    if (pos_in_block < block) {
                        ans += block;
                    }
                    cur_skill = opp_skill ^ cur_xor;
                    cur_size += block;
                    cur_xor ^= opp_xor;
                    cur = opp_start;
                }
            }
            
            cout << ans << '\n';
        }
    }
    
    return 0;
}
