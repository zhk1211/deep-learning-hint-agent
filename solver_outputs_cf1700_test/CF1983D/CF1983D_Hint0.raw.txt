#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n), b(n);
        for (int i = 0; i < n; ++i) cin >> a[i];
        for (int i = 0; i < n; ++i) cin >> b[i];

        // Check if multisets of elements are equal
        vector<int> sa = a, sb = b;
        sort(sa.begin(), sa.end());
        sort(sb.begin(), sb.end());
        if (sa != sb) {
            cout << "NO\n";
            continue;
        }

        // Map value to index in b
        unordered_map<int, int> pos_b;
        for (int i = 0; i < n; ++i) {
            pos_b[b[i]] = i;
        }

        // Build permutation p: p[i] = position of a[i] in b
        vector<int> p(n);
        for (int i = 0; i < n; ++i) {
            p[i] = pos_b[a[i]];
        }

        // Check parity of inversions in p
        // Since we can only swap equal length intervals in both arrays,
        // the parity of the permutation p must be even (i.e., p must be an even permutation)
        // because each operation swaps two elements in a and two elements in b,
        // which changes the parity of p by (-1)*(-1) = +1? Actually:
        // Swapping a_l and a_r changes parity of p by -1.
        // Swapping b_p and b_q changes parity of p by -1.
        // Total change = (-1)*(-1) = +1, so parity of p is invariant.
        // Initially p is some permutation. We want p to become identity (even parity).
        // So p must have even parity.
        // Count inversions modulo 2.
        vector<bool> vis(n, false);
        int parity = 0;
        for (int i = 0; i < n; ++i) {
            if (!vis[i]) {
                int cycle_len = 0;
                int j = i;
                while (!vis[j]) {
                    vis[j] = true;
                    j = p[j];
                    ++cycle_len;
                }
                parity ^= ((cycle_len - 1) & 1);
            }
        }
        cout << (parity == 0 ? "YES" : "NO") << '\n';
    }
    return 0;
}
