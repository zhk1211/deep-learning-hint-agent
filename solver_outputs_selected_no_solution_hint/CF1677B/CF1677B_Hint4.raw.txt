#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        string s;
        cin >> s;
        int total = n * m;
        
        vector<int> ans(total);
        vector<int> col_good(m, 0);
        int good_cols = 0;
        vector<int> row_has_one(n, 0);
        int good_rows = 0;
        
        int last_one = -1;
        
        for (int i = 0; i < total; ++i) {
            int col = i % m;
            if (s[i] == '1') {
                if (!col_good[col]) {
                    col_good[col] = 1;
                    ++good_cols;
                }
                last_one = i;
            }
            
            int row = i / m;
            if (s[i] == '1' && !row_has_one[row]) {
                row_has_one[row] = 1;
                ++good_rows;
            }
            
            if (last_one != -1 && i - last_one < m) {
                ans[i] = good_cols + good_rows;
            } else {
                ans[i] = good_cols + good_rows;
            }
        }
        
        vector<int> row_good(total, 0);
        vector<int> col_good2(total, 0);
        vector<int> res(total);
        
        vector<int> col_has_one(m, 0);
        int cur_col_good = 0;
        for (int i = 0; i < total; ++i) {
            int col = i % m;
            if (s[i] == '1' && !col_has_one[col]) {
                col_has_one[col] = 1;
                ++cur_col_good;
            }
            col_good2[i] = cur_col_good;
        }
        
        vector<int> last_one_in_window(total, -1);
        int last = -1;
        for (int i = 0; i < total; ++i) {
            if (s[i] == '1') last = i;
            last_one_in_window[i] = last;
        }
        
        vector<int> row_good_prefix(total, 0);
        int cur_row_good = 0;
        vector<int> row_has_one2(n, 0);
        for (int i = 0; i < total; ++i) {
            int row = i / m;
            if (s[i] == '1' && !row_has_one2[row]) {
                row_has_one2[row] = 1;
                ++cur_row_good;
            }
            row_good_prefix[i] = cur_row_good;
        }
        
        for (int i = 0; i < total; ++i) {
            int col_part = col_good2[i];
            int row_part = 0;
            if (i >= m) {
                int prev = i - m;
                if (last_one_in_window[i] > prev) {
                    row_part = row_good_prefix[i];
                } else {
                    row_part = row_good_prefix[prev];
                }
            } else {
                if (last_one_in_window[i] != -1) {
                    row_part = row_good_prefix[i];
                } else {
                    row_part = 0;
                }
            }
            res[i] = col_part + row_part;
        }
        
        for (int i = 0; i < total; ++i) {
            cout << res[i] << " \n"[i == total - 1];
        }
    }
    return 0;
}
