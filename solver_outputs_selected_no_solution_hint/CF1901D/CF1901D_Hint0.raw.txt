```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    
    if (n == 1) {
        cout << a[0] << "\n";
        return 0;
    }
    
    vector<long long> b = a;
    for (int i = 0; i < n; ++i) {
        b[i] += (n - 1 - i);
    }
    
    vector<long long> pref_max(n), suff_max(n);
    pref_max[0] = b[0];
    for (int i = 1; i < n; ++i) {
        pref_max[i] = max(pref_max[i - 1], b[i]);
    }
    suff_max[n - 1] = b[n - 1];
    for (int i = n - 2; i >= 0; --i) {
        suff_max[i] = max(suff_max[i + 1], b[i]);
    }
    
    long long ans = LLONG_MAX;
    for (int i = 0; i < n; ++i) {
        long long cand = a[i];
        if (i > 0) {
            cand = max(cand, pref_max[i - 1] - i);
        }
        if (i < n - 1) {
            cand = max(cand, suff_max[i + 1] - (n - 1 - i));
        }
        ans = min(ans, cand);
    }
    
    cout << ans << "\n";
    return 0;
}
```
