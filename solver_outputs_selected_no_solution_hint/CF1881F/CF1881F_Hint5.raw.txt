#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, k;
    cin >> n >> k;
    vector<int> marked(k);
    vector<bool> is_marked(n + 1, false);
    for (int i = 0; i < k; ++i) {
        cin >> marked[i];
        is_marked[marked[i]] = true;
    }
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < n - 1; ++i) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    if (k == 0) {
        cout << 0 << '\n';
        return;
    }
    // First BFS to find one endpoint of the diameter of marked vertices
    vector<int> dist(n + 1, -1);
    queue<int> q;
    q.push(marked[0]);
    dist[marked[0]] = 0;
    int far = marked[0];
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : adj[u]) {
            if (dist[v] == -1) {
                dist[v] = dist[u] + 1;
                q.push(v);
                if (is_marked[v] && dist[v] > dist[far]) far = v;
            }
        }
    }
    // Second BFS from far to find distances and the other endpoint
    vector<int> d1(n + 1, -1);
    q.push(far);
    d1[far] = 0;
    int far2 = far;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : adj[u]) {
            if (d1[v] == -1) {
                d1[v] = d1[u] + 1;
                q.push(v);
                if (is_marked[v] && d1[v] > d1[far2]) far2 = v;
            }
        }
    }
    // Third BFS from far2
    vector<int> d2(n + 1, -1);
    q.push(far2);
    d2[far2] = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : adj[u]) {
            if (d2[v] == -1) {
                d2[v] = d2[u] + 1;
                q.push(v);
            }
        }
    }
    int ans = INT_MAX;
    for (int i = 1; i <= n; ++i) {
        ans = min(ans, max(d1[i], d2[i]));
    }
    cout << ans << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
