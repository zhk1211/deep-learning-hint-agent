```cpp
#include <bits/stdc++.h>
using namespace std;

int query(int i) {
    cout << "? " << i << endl;
    int x;
    cin >> x;
    return x;
}

int main() {
    int n;
    cin >> n;
    
    int l = 1, r = n;
    while (l < r) {
        int mid = (l + r) / 2;
        int a = query(mid);
        int b = query(mid + 1);
        if (a < b) {
            r = mid;
        } else {
            l = mid + 1;
        }
    }
    cout << "! " << l << endl;
    return 0;
}
```
