#include <bits/stdc++.h>
using namespace std;

bool is_prime(int x) {
    if (x < 2) return false;
    for (int i = 2; i * i <= x; ++i)
        if (x % i == 0) return false;
    return true;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> p(n);
        if (n == 2) {
            cout << "2 1\n";
            continue;
        }
        if (n == 3) {
            cout << "2 1 3\n";
            continue;
        }
        // find a prime P such that P > n and P <= 2n-1
        int P = n + 1;
        while (!is_prime(P)) ++P;
        // p1 = P - (n-1)
        p[0] = P - (n - 1);
        // place n at position 2
        p[1] = n;
        // fill remaining with 1..n-1 except p[0]
        vector<bool> used(n + 1, false);
        used[p[0]] = true;
        used[n] = true;
        int idx = 2;
        for (int x = 1; x <= n - 1; ++x) {
            if (!used[x]) {
                p[idx++] = x;
            }
        }
        for (int i = 0; i < n; ++i) {
            cout << p[i] << " \n"[i == n - 1];
        }
    }
    return 0;
}
