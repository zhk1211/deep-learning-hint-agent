#include <bits/stdc++.h>
using namespace std;

bool is_prime(int x) {
    if (x < 2) return false;
    for (int i = 2; i * i <= x; ++i)
        if (x % i == 0) return false;
    return true;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> p(n);
        iota(p.begin(), p.end(), 1);
        
        // We want many c_i to be prime.
        // c_i = ceil(prefix_sum / i)
        // If we put 2,1 at the beginning, then c_1=2, c_2=2 (both prime).
        // For i>=3, we can try to make c_i = 2 or 3 (both prime) by controlling prefix sums.
        // The hint says: put all primes in the beginning.
        // Let's try: 2, 1, then all primes, then the rest.
        
        vector<int> primes, others;
        for (int i = 3; i <= n; ++i) {
            if (is_prime(i)) primes.push_back(i);
            else others.push_back(i);
        }
        
        vector<int> ans;
        ans.push_back(2);
        ans.push_back(1);
        for (int x : primes) ans.push_back(x);
        for (int x : others) ans.push_back(x);
        
        for (int i = 0; i < n; ++i) {
            cout << ans[i] << " \n"[i == n-1];
        }
    }
    return 0;
}
