#include <bits/stdc++.h>
using namespace std;

const int MOD = 998244353;
const int MAXN = 500000;

long long fact[MAXN + 5], invfact[MAXN + 5];

long long modpow(long long a, long long e) {
    long long res = 1;
    while (e) {
        if (e & 1) res = res * a % MOD;
        a = a * a % MOD;
        e >>= 1;
    }
    return res;
}

void precompute() {
    fact[0] = 1;
    for (int i = 1; i <= MAXN; i++) {
        fact[i] = fact[i-1] * i % MOD;
    }
    invfact[MAXN] = modpow(fact[MAXN], MOD-2);
    for (int i = MAXN; i >= 1; i--) {
        invfact[i-1] = invfact[i] * i % MOD;
    }
}

long long nCr(int n, int r) {
    if (r < 0 || r > n) return 0;
    return fact[n] * invfact[r] % MOD * invfact[n-r] % MOD;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    precompute();
    int t;
    cin >> t;
    while (t--) {
        vector<int> c(26);
        int total = 0;
        int oddCount = 0, evenCount = 0;
        bool possible = true;
        for (int i = 0; i < 26; i++) {
            cin >> c[i];
            total += c[i];
            if (c[i] % 2 == 1) oddCount++;
            else if (c[i] > 0) evenCount++;
            if (c[i] >= 3) possible = false;
        }
        if (!possible) {
            cout << 0 << "\n";
            continue;
        }
        if (oddCount > 1) {
            cout << 0 << "\n";
            continue;
        }
        if (total == 0) {
            cout << 0 << "\n";
            continue;
        }
        int oddPositions = (total + 1) / 2;
        int evenPositions = total / 2;
        long long ways = 1;
        int oddSum = 0;
        for (int i = 0; i < 26; i++) {
            if (c[i] == 2) {
                ways = ways * nCr(oddPositions - oddSum, 1) % MOD;
                oddSum += 1;
                ways = ways * nCr(evenPositions - (i > 0 ? 0 : 0), 1) % MOD;
            } else if (c[i] == 1) {
                if (oddCount == 1 && c[i] % 2 == 1) {
                    ways = ways * nCr(oddPositions - oddSum, 1) % MOD;
                    oddSum += 1;
                } else {
                    ways = ways * nCr(evenPositions - (oddSum > 0 ? 0 : 0), 1) % MOD;
                }
            }
        }
        cout << ways % MOD << "\n";
    }
    return 0;
}
