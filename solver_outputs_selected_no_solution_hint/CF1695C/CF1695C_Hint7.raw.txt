```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<vector<int>> a(n, vector<int>(m));
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < m; ++j) {
                cin >> a[i][j];
            }
        }
        
        if ((n + m) % 2 == 0) {
            cout << "NO\n";
            continue;
        }
        
        vector<vector<int>> mn(n, vector<int>(m));
        vector<vector<int>> mx(n, vector<int>(m));
        
        mn[0][0] = mx[0][0] = a[0][0];
        
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < m; ++j) {
                if (i == 0 && j == 0) continue;
                int cur = a[i][j];
                if (i == 0) {
                    mn[i][j] = mn[i][j-1] + cur;
                    mx[i][j] = mx[i][j-1] + cur;
                } else if (j == 0) {
                    mn[i][j] = mn[i-1][j] + cur;
                    mx[i][j] = mx[i-1][j] + cur;
                } else {
                    mn[i][j] = min(mn[i-1][j], mn[i][j-1]) + cur;
                    mx[i][j] = max(mx[i-1][j], mx[i][j-1]) + cur;
                }
            }
        }
        
        if (mn[n-1][m-1] <= 0 && 0 <= mx[n-1][m-1]) {
            cout << "YES\n";
        } else {
            cout << "NO\n";
        }
    }
    
    return 0;
}
```
