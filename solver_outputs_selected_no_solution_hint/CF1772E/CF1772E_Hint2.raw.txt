#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> p(n);
        for (int i = 0; i < n; ++i) {
            cin >> p[i];
        }
        
        // Find longest prefix that is already sorted ascending
        int pref = 0;
        while (pref < n && p[pref] == pref + 1) {
            ++pref;
        }
        
        // Find longest suffix that is already sorted descending
        int suff = 0;
        while (suff < n && p[n - 1 - suff] == n - suff) {
            ++suff;
        }
        
        // Count elements that are in correct position for ascending
        // but not part of the prefix
        int cnt_first = 0;
        for (int i = pref; i < n - suff; ++i) {
            if (p[i] == i + 1) {
                ++cnt_first;
            }
        }
        
        // Count elements that are in correct position for descending
        // but not part of the suffix
        int cnt_second = 0;
        for (int i = pref; i < n - suff; ++i) {
            if (p[i] == n - i) {
                ++cnt_second;
            }
        }
        
        // The middle segment length
        int mid = n - pref - suff;
        
        // Determine winner based on counts
        if (pref + suff >= n) {
            // Already sorted? But problem says not sorted initially.
            // This case shouldn't happen per constraints.
            cout << "Tie\n";
        } else if (cnt_first + pref == n) {
            // Already ascending? Not possible per constraints.
            cout << "Tie\n";
        } else if (cnt_second + suff == n) {
            // Already descending? Not possible per constraints.
            cout << "Tie\n";
        } else {
            // First player wants ascending, second wants descending
            // First player needs to fix elements not in ascending position
            // Second player needs to fix elements not in descending position
            int need_first = mid - cnt_first;
            int need_second = mid - cnt_second;
            
            if (need_first <= need_second) {
                if (need_first == 0) {
                    cout << "First\n";
                } else if (need_first < need_second) {
                    cout << "First\n";
                } else {
                    // need_first == need_second
                    if (cnt_first > cnt_second) {
                        cout << "First\n";
                    } else if (cnt_first < cnt_second) {
                        cout << "Second\n";
                    } else {
                        cout << "Tie\n";
                    }
                }
            } else {
                if (need_second == 0) {
                    cout << "Second\n";
                } else {
                    cout << "Second\n";
                }
            }
        }
    }
    
    return 0;
}
