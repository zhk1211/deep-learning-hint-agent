```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> p(n);
        for (int i = 0; i < n; ++i) {
            cin >> p[i];
        }

        // Find the longest prefix that is already sorted ascending
        int pref = 0;
        while (pref < n && p[pref] == pref + 1) {
            ++pref;
        }

        // Find the longest suffix that is already sorted descending
        int suff = 0;
        while (suff < n && p[n - 1 - suff] == n - suff) {
            ++suff;
        }

        // Count elements that are in correct position for ascending order
        int cnt_asc = 0;
        for (int i = 0; i < n; ++i) {
            if (p[i] == i + 1) {
                ++cnt_asc;
            }
        }

        // Count elements that are in correct position for descending order
        int cnt_desc = 0;
        for (int i = 0; i < n; ++i) {
            if (p[i] == n - i) {
                ++cnt_desc;
            }
        }

        // The number of elements that are not in correct position for ascending
        int wrong_asc = n - cnt_asc;
        // The number of elements that are not in correct position for descending
        int wrong_desc = n - cnt_desc;

        // The first player needs to fix all wrong_asc elements.
        // The second player needs to fix all wrong_desc elements.
        // However, they can interfere with each other's progress.

        // The key insight from hints: the game reduces to comparing
        // the number of "red" elements that are smaller than all blue elements, etc.
        // But a simpler known solution for this problem:
        // Let a = number of elements not in correct position for ascending
        // Let b = number of elements not in correct position for descending
        // Let c = number of elements that are wrong for both (intersection)
        // Actually, we need to consider the sets of positions that are wrong for each.

        // Positions where p[i] != i+1 (wrong for asc)
        // Positions where p[i] != n-i (wrong for desc)
        // Let A = set of indices where p[i] != i+1
        // Let B = set of indices where p[i] != n-i
        // Let both = |A ∩ B|, onlyA = |A \ B|, onlyB = |B \ A|

        vector<bool> wrongA(n, false), wrongB(n, false);
        for (int i = 0; i < n; ++i) {
            if (p[i] != i + 1) wrongA[i] = true;
            if (p[i] != n - i) wrongB[i] = true;
        }

        int both = 0, onlyA = 0, onlyB = 0;
        for (int i = 0; i < n; ++i) {
            if (wrongA[i] && wrongB[i]) both++;
            else if (wrongA[i]) onlyA++;
            else if (wrongB[i]) onlyB++;
        }

        // First player wins if they can fix all their wrong positions before second player fixes theirs.
        // The game is sequential: players can color elements blue and then rearrange blue elements.
        // The optimal strategy: each player colors elements that are wrong for them but not for the opponent,
        // then they fight over the "both" elements.

        // Let's simulate the race:
        // Turn order: First, Second, First, Second, ...
        // On a turn, a player can either color a red element blue, or rearrange blue elements to win if possible.
        // They can also skip.

        // The known solution from editorial:
        // If onlyA + both <= onlyB: First wins (First can fix all their problems before Second can win)
        // If onlyB + both < onlyA: Second wins
        // Else: Tie

        // Wait, check sample:
        // Sample 1: 1 2 4 3
        // wrongA: positions 2,3 (0-indexed: 2,3) -> p[2]=4!=3, p[3]=3!=4
        // wrongB: p[0]=1!=4, p[1]=2!=3, p[2]=4!=2, p[3]=3!=1 -> all wrong? Actually descending: [4,3,2,1]
        // p = [1,2,4,3]
        // i=0: p[0]=1, n-i=4 -> wrongB
        // i=1: p[1]=2, n-i=3 -> wrongB
        // i=2: p[2]=4, n-i=2 -> wrongB? 4!=2 yes
        // i=3: p[3]=3, n-i=1 -> wrongB
        // So wrongB all true.
        // onlyA = 0, onlyB = 2? Actually both = 2 (positions 2,3), onlyB = 2 (positions 0,1)
        // onlyA + both = 2, onlyB = 2 -> condition onlyA + both <= onlyB gives 2 <= 2 true -> First wins. Correct.

        // Sample 2: 2 3 1
        // n=3, p=[2,3,1]
        // wrongA: p[0]=2!=1, p[1]=3!=2, p[2]=1!=3 -> all wrong
        // wrongB: p[0]=2!=3, p[1]=3!=2, p[2]=1!=1? p[2]=1, n-2=1 -> correct. So wrongB: [true, true, false]
        // both: positions 0,1 -> 2
        // onlyA: position 2 -> 1
        // onlyB: 0
        // onlyA + both = 3, onlyB = 0 -> 3 <= 0 false
        // onlyB + both = 2, onlyA = 1 -> 2 < 1 false
        // So Tie. Correct.

        // Sample 3: 3 4 5 2 1
        // n=5
        // p = [3,4,5,2,1]
        // wrongA: p[0]=3!=1, p[1]=4!=2, p[2]=5!=3, p[3]=2!=4, p[4]=1!=5 -> all wrong
        // wrongB: p[0]=3!=5, p[1]=4!=4? p[1]=4, n-1=4 -> correct. p[2]=5!=3, p[3]=2!=2? p[3]=2, n-3=2 -> correct. p[4]=1!=1? p[4]=1, n-4=1 -> correct.
        // So wrongB: [true, false, true, false, false]
        // both: positions 0,2 -> 2
        // onlyA: positions 1,3,4 -> 3
        // onlyB: 0
        // onlyA + both = 5, onlyB = 0 -> false
        // onlyB + both = 2, onlyA = 3 -> 2 < 3 true -> Second wins. Correct.

        // Sample 4: 1 5 6 3 2 4
        // n=6
        // p = [1,5,6,3,2,4]
        // wrongA: p[0]=1 ok, p[1]=5!=2, p[2]=6!=3, p[3]=3!=4, p[4]=2!=5, p[5]=4!=6 -> wrongA: [false, true, true, true, true, true]
        // wrongB: p[0]=1!=6, p[1]=5!=5? p[1]=5, n-1=5 -> correct. p[2]=6!=4, p[3]=3!=3? p[3]=3, n-3=3 -> correct. p[4]=2!=2? p[4]=2, n-4=2 -> correct. p[5]=4!=1 -> wrongB: [true, false, true, false, false, true]
        // both: positions 0,2,5 -> 3
        // onlyA: positions 1,3,4 -> 3
        // onlyB: 0
        // onlyA + both = 6, onlyB = 0 -> false
        // onlyB + both = 3, onlyA = 3 -> 3 < 3 false -> Tie. Correct.

        if (onlyA + both <= onlyB) {
            cout << "First\n";
        } else if (onlyB + both < onlyA) {
            cout << "Second\n";
        } else {
            cout << "Tie\n";
        }
    }
    return 0;
}
```
