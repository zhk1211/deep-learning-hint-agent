```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }

    // dp[i][j] = min coins for prefix up to i, where j indicates state of the last element:
    // j = 0: last element is red and its value is 0 (cannot be used to expand)
    // j = 1: last element is red and its value is 1 (can expand once)
    // j = 2: last element is red and its value is 2 (can expand twice)
    // j = 3: last element is blue (not yet painted)
    const int INF = 1e9;
    vector<array<int, 4>> dp(n + 1);
    dp[0] = {0, INF, INF, INF};

    for (int i = 0; i < n; ++i) {
        dp[i + 1].fill(INF);
        int val = a[i];

        // Transition from state 0 (red 0)
        if (dp[i][0] != INF) {
            // Option: paint current blue with coin
            if (val == 0) dp[i + 1][0] = min(dp[i + 1][0], dp[i][0] + 1);
            else if (val == 1) dp[i + 1][1] = min(dp[i + 1][1], dp[i][0] + 1);
            else if (val == 2) dp[i + 1][2] = min(dp[i + 1][2], dp[i][0] + 1);
            // Option: leave current blue
            dp[i + 1][3] = min(dp[i + 1][3], dp[i][0]);
        }

        // Transition from state 1 (red 1)
        if (dp[i][1] != INF) {
            // Use the red 1 to paint current blue (decrease to 0)
            if (val == 0) dp[i + 1][0] = min(dp[i + 1][0], dp[i][1]);
            else if (val == 1) dp[i + 1][1] = min(dp[i + 1][1], dp[i][1]);
            else if (val == 2) dp[i + 1][2] = min(dp[i + 1][2], dp[i][1]);
            // Or paint with coin
            if (val == 0) dp[i + 1][0] = min(dp[i + 1][0], dp[i][1] + 1);
            else if (val == 1) dp[i + 1][1] = min(dp[i + 1][1], dp[i][1] + 1);
            else if (val == 2) dp[i + 1][2] = min(dp[i + 1][2], dp[i][1] + 1);
            // Leave blue
            dp[i + 1][3] = min(dp[i + 1][3], dp[i][1]);
        }

        // Transition from state 2 (red 2)
        if (dp[i][2] != INF) {
            // Use red 2 to paint current (decrease to 1)
            if (val == 0) dp[i + 1][0] = min(dp[i + 1][0], dp[i][2]);
            else if (val == 1) dp[i + 1][1] = min(dp[i + 1][1], dp[i][2]);
            else if (val == 2) dp[i + 1][2] = min(dp[i + 1][2], dp[i][2]);
            // Or paint with coin
            if (val == 0) dp[i + 1][0] = min(dp[i + 1][0], dp[i][2] + 1);
            else if (val == 1) dp[i + 1][1] = min(dp[i + 1][1], dp[i][2] + 1);
            else if (val == 2) dp[i + 1][2] = min(dp[i + 1][2], dp[i][2] + 1);
            // Leave blue
            dp[i + 1][3] = min(dp[i + 1][3], dp[i][2]);
        }

        // Transition from state 3 (blue)
        if (dp[i][3] != INF) {
            // Must paint current with coin (since previous is blue, no red adjacent to use)
            if (val == 0) dp[i + 1][0] = min(dp[i + 1][0], dp[i][3] + 1);
            else if (val == 1) dp[i + 1][1] = min(dp[i + 1][1], dp[i][3] + 1);
            else if (val == 2) dp[i + 1][2] = min(dp[i + 1][2], dp[i][3] + 1);
            // Also can leave current blue
            dp[i + 1][3] = min(dp[i + 1][3], dp[i][3]);
        }
    }

    int ans = min({dp[n][0], dp[n][1], dp[n][2]});
    cout << ans << '\n';

    return 0;
}
```
