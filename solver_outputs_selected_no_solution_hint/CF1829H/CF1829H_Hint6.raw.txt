#include <bits/stdc++.h>
using namespace std;

const int MOD = 1e9 + 7;

void solve() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    
    // dp[mask] = number of subsequences with AND exactly mask
    vector<long long> dp(64, 0);
    
    for (int x : a) {
        vector<long long> new_dp = dp;
        // subsequence consisting of only x
        new_dp[x] = (new_dp[x] + 1) % MOD;
        // extend existing subsequences
        for (int mask = 0; mask < 64; ++mask) {
            if (dp[mask] > 0) {
                int new_mask = mask & x;
                new_dp[new_mask] = (new_dp[new_mask] + dp[mask]) % MOD;
            }
        }
        dp = move(new_dp);
    }
    
    long long ans = 0;
    for (int mask = 0; mask < 64; ++mask) {
        if (__builtin_popcount(mask) == k) {
            ans = (ans + dp[mask]) % MOD;
        }
    }
    cout << ans << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
