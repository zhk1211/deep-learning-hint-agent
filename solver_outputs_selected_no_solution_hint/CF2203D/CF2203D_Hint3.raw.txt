#include <bits/stdc++.h>
using namespace std;

const int MAXV = 2000005;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<int> a(n), b(m);
        for (int i = 0; i < n; ++i) cin >> a[i];
        for (int i = 0; i < m; ++i) cin >> b[i];

        int max_val = 0;
        for (int x : a) max_val = max(max_val, x);
        for (int x : b) max_val = max(max_val, x);

        vector<int> cnt_a(max_val + 1, 0);
        for (int x : a) cnt_a[x]++;

        vector<int> cnt_b(max_val + 1, 0);
        for (int x : b) cnt_b[x]++;

        vector<int> good(max_val + 1, 0);
        for (int i = 1; i <= max_val; ++i) {
            if (cnt_a[i] == 0) continue;
            for (int j = i; j <= max_val; j += i) {
                good[j] += cnt_a[i];
            }
        }

        vector<int> moves(max_val + 1, 0);
        for (int i = 1; i <= max_val; ++i) {
            if (cnt_b[i] == 0) continue;
            if (good[i] > 0) {
                moves[i] = 1;
            } else {
                moves[i] = 0;
            }
        }

        int total_moves = 0;
        for (int i = 1; i <= max_val; ++i) {
            total_moves += moves[i] * cnt_b[i];
        }

        if (total_moves % 2 == 1) {
            cout << "Alice\n";
        } else {
            cout << "Bob\n";
        }
    }
    return 0;
}
