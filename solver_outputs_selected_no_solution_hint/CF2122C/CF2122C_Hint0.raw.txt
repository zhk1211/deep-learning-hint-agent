#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<pair<long long, int>> a(n), b(n);
        for (int i = 0; i < n; ++i) {
            long long x, y;
            cin >> x >> y;
            a[i] = {x + y, i + 1};
            b[i] = {x - y, i + 1};
        }
        sort(a.begin(), a.end());
        sort(b.begin(), b.end());
        vector<int> used(n + 1, 0);
        vector<pair<int, int>> ans;
        int l = 0, r = n - 1;
        while (l < r) {
            int u = a[l].second, v = a[r].second;
            if (!used[u] && !used[v]) {
                used[u] = used[v] = 1;
                ans.emplace_back(u, v);
            }
            ++l; --r;
        }
        if ((int)ans.size() < n / 2) {
            l = 0, r = n - 1;
            while (l < r && (int)ans.size() < n / 2) {
                int u = b[l].second, v = b[r].second;
                if (!used[u] && !used[v]) {
                    used[u] = used[v] = 1;
                    ans.emplace_back(u, v);
                }
                ++l; --r;
            }
        }
        for (auto &p : ans) {
            cout << p.first << ' ' << p.second << '\n';
        }
    }
    return 0;
}
