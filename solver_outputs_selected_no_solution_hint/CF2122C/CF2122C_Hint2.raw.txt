```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<pair<int, int>> pts(n);
        vector<int> idx(n);
        for (int i = 0; i < n; ++i) {
            cin >> pts[i].first >> pts[i].second;
            idx[i] = i + 1;
        }
        sort(idx.begin(), idx.end(), [&](int i, int j) {
            return pts[i - 1].first + pts[i - 1].second < pts[j - 1].first + pts[j - 1].second;
        });
        for (int i = 0; i < n / 2; ++i) {
            cout << idx[i] << ' ' << idx[n - 1 - i] << '\n';
        }
    }
    return 0;
}
```
