```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }

        vector<int> sorted_a = a;
        sort(sorted_a.begin(), sorted_a.end());

        int max_wins = 0;
        long long sum = 0;
        for (int i = 0; i < n; ++i) {
            if (sum + sorted_a[i] <= m) {
                sum += sorted_a[i];
                max_wins++;
            } else {
                break;
            }
        }

        if (max_wins == n) {
            cout << 1 << '\n';
            continue;
        }

        int place = n - max_wins + 1;

        if (max_wins > 0 && a[max_wins] <= m) {
            long long sum_with = 0;
            int cnt = 0;
            bool used = false;
            for (int i = 0; i < n; ++i) {
                if (i == max_wins) {
                    if (sum_with + a[i] <= m) {
                        sum_with += a[i];
                        cnt++;
                        used = true;
                    }
                } else {
                    if (sum_with + sorted_a[cnt] <= m) {
                        sum_with += sorted_a[cnt];
                        cnt++;
                    } else {
                        break;
                    }
                }
            }
            if (used && cnt >= max_wins + 1) {
                place = n - max_wins;
            }
        }

        cout << place << '\n';
    }
    return 0;
}
```
