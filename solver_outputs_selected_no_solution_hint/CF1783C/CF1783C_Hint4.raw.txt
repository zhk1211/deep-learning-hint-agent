```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }
        
        vector<int> sorted_a = a;
        sort(sorted_a.begin(), sorted_a.end());
        
        int max_wins = 0;
        long long sum = 0;
        for (int i = 0; i < n; ++i) {
            if (sum + sorted_a[i] <= m) {
                sum += sorted_a[i];
                max_wins++;
            } else {
                break;
            }
        }
        
        if (max_wins == n) {
            cout << 1 << "\n";
            continue;
        }
        
        if (max_wins == 0) {
            cout << n + 1 << "\n";
            continue;
        }
        
        int idx = max_wins;
        if (a[idx] <= sorted_a[max_wins - 1] || sum - sorted_a[max_wins - 1] + a[idx] <= m) {
            cout << n - max_wins + 1 << "\n";
        } else {
            cout << n - max_wins + 2 << "\n";
        }
    }
    
    return 0;
}
```
