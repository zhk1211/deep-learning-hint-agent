#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    long long m;
    cin >> n >> m;
    vector<long long> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }

    sort(a.begin(), a.end());

    long long low = 0, high = 2e14;
    long long ans = high;

    while (low <= high) {
        long long mid = (low + high) / 2;
        long long disk = 0;
        long long time_passed = 0;
        bool ok = true;

        for (int i = 0; i < n; ++i) {
            long long s = a[i];
            if (disk + s <= m) {
                disk += s;
                time_passed += s;
            } else {
                long long need = disk + s - m;
                if (time_passed + need > mid) {
                    ok = false;
                    break;
                }
                time_passed += need;
                disk = m;
                disk -= (disk + s - m);
                disk = m;
                time_passed += s;
                disk = s;
            }
            if (time_passed > mid) {
                ok = false;
                break;
            }
            time_passed += 1;
            disk -= s;
            if (disk < 0) disk = 0;
        }

        if (ok) {
            ans = mid;
            high = mid - 1;
        } else {
            low = mid + 1;
        }
    }

    cout << ans << '\n';
    return 0;
}
