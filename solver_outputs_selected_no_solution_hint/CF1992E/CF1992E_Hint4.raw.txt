            // Let's consider if there is any other case where d=0? n_int = len. For n from 1 to 9, len=1, so n_int=1 => n=1. For n=10..99, len=2, n_int would need to be 2, but n>=10, so no. So only n=1 gives d=0.

            // So the solution is complete.

            // Let's write the final code.
        }

        // Output
        cout << ans.size() << '\n';
        for (auto &p : ans) {
            cout << p.first << ' ' << p.second << '\n';
        }
    }

    return 0;
