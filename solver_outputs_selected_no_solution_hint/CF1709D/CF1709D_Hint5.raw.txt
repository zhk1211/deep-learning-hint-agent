```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXM = 200005;
int st[MAXM][18]; // sparse table for max
int lg[MAXM];

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;
    vector<int> a(m + 1);
    for (int i = 1; i <= m; ++i) {
        cin >> a[i];
    }

    // Precompute log2
    lg[1] = 0;
    for (int i = 2; i <= m; ++i) {
        lg[i] = lg[i / 2] + 1;
    }

    // Build sparse table for range maximum query on a[1..m]
    for (int i = 1; i <= m; ++i) {
        st[i][0] = a[i];
    }
    for (int j = 1; (1 << j) <= m; ++j) {
        for (int i = 1; i + (1 << j) - 1 <= m; ++i) {
            st[i][j] = max(st[i][j - 1], st[i + (1 << (j - 1))][j - 1]);
        }
    }

    auto range_max = [&](int l, int r) {
        int j = lg[r - l + 1];
        return max(st[l][j], st[r - (1 << j) + 1][j]);
    };

    int q;
    cin >> q;
    while (q--) {
        int xs, ys, xf, yf, k;
        cin >> xs >> ys >> xf >> yf >> k;

        if (abs(ys - yf) % k != 0) {
            cout << "NO\n";
            continue;
        }
        if (abs(xs - xf) % k != 0) {
            cout << "NO\n";
            continue;
        }

        int max_reachable_row = n - ((n - xs) % k);
        int l = min(ys, yf);
        int r = max(ys, yf);
        int max_blocked = range_max(l, r);

        if (max_reachable_row > max_blocked) {
            cout << "YES\n";
        } else {
            cout << "NO\n";
        }
    }

    return 0;
}
```
