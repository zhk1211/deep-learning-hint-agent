#include <bits/stdc++.h>
using namespace std;
using ll = long long;

void solve() {
    int n, q;
    cin >> n >> q;
    vector<ll> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    vector<ll> pref(n + 1, 0);
    for (int i = 0; i < n; ++i) {
        pref[i + 1] = pref[i] + a[i];
    }
    ll total = pref[n];
    auto get_sum = [&](ll idx) -> ll {
        if (idx <= 0) return 0;
        ll block = (idx - 1) / n;
        ll pos = (idx - 1) % n;
        ll sum = block * total;
        ll rem = pos + 1;
        if (block + rem <= n) {
            sum += pref[block + rem] - pref[block];
        } else {
            sum += pref[n] - pref[block];
            sum += pref[block + rem - n];
        }
        return sum;
    };
    while (q--) {
        ll l, r;
        cin >> l >> r;
        cout << get_sum(r) - get_sum(l - 1) << '\n';
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
