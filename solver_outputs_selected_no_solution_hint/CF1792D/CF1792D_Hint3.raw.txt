#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<vector<int>> a(n, vector<int>(m));
        vector<vector<int>> pos(n, vector<int>(m + 1));
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < m; ++j) {
                cin >> a[i][j];
                pos[i][a[i][j]] = j;
            }
        }
        vector<int> max_beauty(1 << m, 0);
        for (int i = 0; i < n; ++i) {
            int mask = 0;
            for (int k = 1; k <= m; ++k) {
                if (pos[i][k] == k - 1) {
                    mask |= (1 << (k - 1));
                } else {
                    break;
                }
            }
            max_beauty[mask] = max(max_beauty[mask], __builtin_popcount(mask));
        }
        for (int mask = (1 << m) - 1; mask >= 0; --mask) {
            for (int bit = 0; bit < m; ++bit) {
                if (mask & (1 << bit)) {
                    max_beauty[mask ^ (1 << bit)] = max(max_beauty[mask ^ (1 << bit)], max_beauty[mask]);
                }
            }
        }
        for (int i = 0; i < n; ++i) {
            int mask = 0;
            for (int k = 1; k <= m; ++k) {
                int val = a[i][k - 1];
                if (pos[i][val] == k - 1) {
                    mask |= (1 << (k - 1));
                } else {
                    break;
                }
            }
            cout << max_beauty[mask] << " \n"[i == n - 1];
        }
    }
    return 0;
}
