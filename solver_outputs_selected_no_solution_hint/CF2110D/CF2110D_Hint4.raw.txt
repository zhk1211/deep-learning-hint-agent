```cpp
#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll INF = 1e18;

struct Edge {
    int to;
    ll w;
};

void solve() {
    int n, m;
    cin >> n >> m;
    vector<ll> b(n);
    for (int i = 0; i < n; ++i) cin >> b[i];
    vector<vector<Edge>> g(n);
    for (int i = 0; i < m; ++i) {
        int s, t; ll w;
        cin >> s >> t >> w;
        --s; --t;
        g[s].push_back({t, w});
    }
    
    vector<ll> dp(n, INF);
    dp[0] = 0;
    priority_queue<pair<ll, int>, vector<pair<ll, int>>, greater<>> pq;
    pq.push({0, 0});
    
    while (!pq.empty()) {
        auto [d, v] = pq.top(); pq.pop();
        if (d != dp[v]) continue;
        ll cur = max(d, b[v]);
        for (auto &e : g[v]) {
            ll need = max(cur, e.w);
            if (need < dp[e.to]) {
                dp[e.to] = need;
                pq.push({need, e.to});
            }
        }
    }
    
    if (dp[n-1] == INF) cout << "-1\n";
    else cout << dp[n-1] << "\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) solve();
    return 0;
}
```
