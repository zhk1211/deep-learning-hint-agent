```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;
const ll INF = 1e18;

void solve() {
    int n, m;
    cin >> n >> m;
    vector<ll> b(n);
    for (int i = 0; i < n; ++i) {
        cin >> b[i];
    }
    vector<vector<pair<int, ll>>> g(n);
    for (int i = 0; i < m; ++i) {
        int u, v;
        ll w;
        cin >> u >> v >> w;
        --u; --v;
        g[u].emplace_back(v, w);
    }
    
    vector<ll> dp(n, INF);
    dp[0] = 0;
    vector<int> topo;
    vector<int> indeg(n, 0);
    for (int u = 0; u < n; ++u) {
        for (auto &[v, w] : g[u]) {
            indeg[v]++;
        }
    }
    queue<int> q;
    for (int i = 0; i < n; ++i) {
        if (indeg[i] == 0) q.push(i);
    }
    while (!q.empty()) {
        int u = q.front(); q.pop();
        topo.push_back(u);
        for (auto &[v, w] : g[u]) {
            if (--indeg[v] == 0) q.push(v);
        }
    }
    
    for (int u : topo) {
        if (dp[u] == INF) continue;
        ll cur = dp[u];
        if (cur < b[u]) {
            cur = b[u];
        }
        for (auto &[v, w] : g[u]) {
            ll need = max(cur, w);
            if (need < dp[v]) {
                dp[v] = need;
            }
        }
    }
    
    if (dp[n-1] == INF) {
        cout << -1 << '\n';
    } else {
        cout << dp[n-1] << '\n';
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
```
