#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const ll INF = 1e18;

void solve() {
    int n, m;
    cin >> n >> m;
    vector<vector<pair<int, ll>>> adj(n + 1);
    for (int i = 0; i < m; ++i) {
        int a, b;
        ll d;
        cin >> a >> b >> d;
        // a is d meters in front of b => pos[a] - pos[b] = d
        // pos[a] <= pos[b] + d
        // pos[b] <= pos[a] - d
        adj[b].push_back({a, d});
        adj[a].push_back({b, -d});
    }
    vector<ll> dist(n + 1, INF);
    vector<bool> in_queue(n + 1, false);
    vector<int> cnt(n + 1, 0);
    queue<int> q;
    for (int i = 1; i <= n; ++i) {
        if (dist[i] == INF) {
            dist[i] = 0;
            q.push(i);
            in_queue[i] = true;
            while (!q.empty()) {
                int u = q.front();
                q.pop();
                in_queue[u] = false;
                for (auto &[v, w] : adj[u]) {
                    if (dist[v] > dist[u] + w) {
                        dist[v] = dist[u] + w;
                        if (!in_queue[v]) {
                            q.push(v);
                            in_queue[v] = true;
                            cnt[v]++;
                            if (cnt[v] > n) {
                                cout << "NO\n";
                                return;
                            }
                        }
                    }
                }
            }
        }
    }
    cout << "YES\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
