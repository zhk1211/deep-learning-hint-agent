```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }
        
        int ans = n;
        int i = n - 2;
        while (i >= 0 && a[i] < a[i + 1]) {
            --i;
        }
        if (i < 0) {
            ans = 1;
        } else {
            ans = i + 2;
        }
        cout << ans << '\n';
    }
    return 0;
}
```
