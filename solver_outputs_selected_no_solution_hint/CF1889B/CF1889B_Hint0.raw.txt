#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

struct DSU {
    vector<int> parent;
    vector<ll> sum;
    DSU(int n, const vector<ll>& a) {
        parent.resize(n);
        sum.resize(n);
        for (int i = 0; i < n; i++) {
            parent[i] = i;
            sum[i] = a[i];
        }
    }
    int find(int x) {
        return parent[x] == x ? x : parent[x] = find(parent[x]);
    }
    void unite(int x, int y) {
        x = find(x); y = find(y);
        if (x == y) return;
        if (sum[x] < sum[y]) swap(x, y);
        parent[y] = x;
        sum[x] += sum[y];
    }
};

void solve() {
    int n; ll c;
    cin >> n >> c;
    vector<ll> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    
    vector<pair<ll, int>> comps;
    for (int i = 1; i < n; i++) {
        comps.push_back({a[i] - c * (i + 1) * 1, i});
    }
    sort(comps.begin(), comps.end(), greater<pair<ll, int>>());
    
    DSU dsu(n, a);
    ll cur_sum = a[0];
    for (auto &p : comps) {
        int idx = p.second;
        if (dsu.find(idx) == dsu.find(0)) continue;
        if (cur_sum + a[idx] >= c * (idx + 1) * 1) {
            dsu.unite(0, idx);
            cur_sum = dsu.sum[dsu.find(0)];
        } else {
            cout << "NO\n";
            return;
        }
    }
    cout << "YES\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
