#include <bits/stdc++.h>
using namespace std;
using ll = long long;

void solve() {
    int n;
    ll c;
    cin >> n >> c;
    vector<ll> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    vector<int> idx(n);
    iota(idx.begin(), idx.end(), 0);
    sort(idx.begin(), idx.end(), [&](int i, int j) {
        return a[i] < a[j];
    });
    ll sum = a[idx[0]];
    for (int k = 1; k < n; ++k) {
        int i = idx[k];
        if (sum + a[i] >= (ll)(i + 1) * (idx[0] + 1) * c) {
            sum += a[i];
        } else {
            cout << "NO\n";
            return;
        }
    }
    cout << "YES\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
