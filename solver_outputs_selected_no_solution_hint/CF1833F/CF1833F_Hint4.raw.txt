#include <bits/stdc++.h>
using namespace std;

const int MOD = 1e9 + 7;

long long modpow(long long a, long long e) {
    long long res = 1;
    while (e) {
        if (e & 1) res = res * a % MOD;
        a = a * a % MOD;
        e >>= 1;
    }
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }
        sort(a.begin(), a.end());

        vector<pair<int, int>> freq;
        for (int x : a) {
            if (freq.empty() || freq.back().first != x) {
                freq.emplace_back(x, 1);
            } else {
                freq.back().second++;
            }
        }

        int k = freq.size();
        vector<long long> pref(k + 1, 1);
        for (int i = 0; i < k; ++i) {
            pref[i + 1] = pref[i] * freq[i].second % MOD;
        }

        long long ans = 0;
        int r = 0;
        for (int l = 0; l < k; ++l) {
            while (r < k && freq[r].first - freq[l].first < m) {
                ++r;
            }
            if (r - l >= m) {
                long long ways = pref[r] * modpow(pref[l], MOD - 2) % MOD;
                ans = (ans + ways) % MOD;
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
